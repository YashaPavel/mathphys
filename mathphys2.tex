\section*{\S 0. Разбиение единицы}
\addcontentsline{toc}{section}{\S 0. Разбиение единицы}
С прошлого семетсра у нас остался долг по обсуждению сюжета про разбиение единицы. В этом разделе мы разберемся с этим вопросом.\\
Рассмотрим функцию
\[
\omega(x)=
\begin{cases}
C\exp\!\left(-\dfrac{1}{1-|x|^2}\right), & |x|<1,\\[2mm]
0, & |x|\ge 1,
\end{cases}
\]
где нормировочная константа $C$ выбрана так, что
\[
\omega(x)>0,
\qquad
\int_{\mathbb R^n}\omega(x)\,\dd x=1.
\]
Тогда
\[
\omega\in C_0^\infty(\mathbb R^n),
\qquad
\operatorname{supp}\omega=\overline{B_1(0)}.
\]

Для любого $\rho>0$ заведём такую функцию:
\[
\omega_\rho(x)=\frac{1}{\rho^n}\omega\!\left(\frac{x}{\rho}\right).
\]
Тогда
\[
\omega_\rho\in C_0^\infty(\mathbb R^n),
\qquad
\omega_\rho>0,
\qquad
\operatorname{supp}\omega_\rho=\overline{B_\rho(0)},
\qquad
\int_{\mathbb R^n}\omega_\rho(x)\,\dd x=1.
\]
Делим на $\rho$ мы для того, чтобы интеграл оставался равным единице (залезет в дифференциал).
\begin{mylemma}
Пусть $\Omega$ и $U$ --- открытые множества и
$
\overline\Omega\subset U.
$
Тогда существует функция $\eta$, 
\[
\eta\in C_0^\infty(\mathbb R^n),
\qquad
0\le \eta(x)\le 1\quad \forall x,
\]
такая, что
\[
\eta\big|_{\overline\Omega}=1,
\qquad
\operatorname{supp}\eta\subset U.
\]
\end{mylemma}

\begin{myproof}
Положим
\[
\rho:=\frac14\operatorname{dist}(\overline\Omega,\partial U)>0.
\]
Введём множество
\[
A:=\left\{x\in\mathbb R^n:\operatorname{dist}(x,\overline\Omega)<2\rho\right\}.
\]
Тогда $A$ открыто и
$
\overline\Omega\subset A,
\qquad
\overline A\subset U.
$

Положим
\[
\eta(x):=\int_A\omega_\rho(x-y)\,\dd y
=(\omega_\rho*\chi_A)(x).
\]
$\chi_A$ - характеристическая функция на множестве A.
Имеем
\[
\eta\in C^\infty(\text{можно дифф. по х}),
\qquad
0\le \eta(x)\le 1(\text{1 равно на всем мн-ве}).
\]
\begin{figure}[H]
        \centering
        \includegraphics[width=0.7\linewidth]{pics/10.png}
        \caption{}
\end{figure}

Если $x\in\overline\Omega$, то из определения $A$
\[
B_\rho(x)\subset A \Rightarrow
\eta(x)
=
\int_{B_\rho(x)}\omega_\rho(x-y)\,\dd y
=1 (\text{тк снаружи шара равна 0, а весь шар внутри A})
\]

Если $x\notin U$, то (учитывая то, что $\operatorname{dist}(\overline{A}, \partial U)\leq2\rho$)
\[
B_\rho(x)\cap A=\varnothing \Rightarrow
\eta(x)=0.
\]
\end{myproof}

\begin{mylemma}
Пусть $\Omega$ --- открытая ограниченная область,
\[
\overline\Omega\subset\bigcup_{j=1}^{N}O_j,
\qquad O_j\text{ --- открытые}.
\]
Пусть $Q$ - куб такой, что $\overline{\bigcup_{j=1}^{N}O_j}\subset Q$. Тогда существуют открытые множества
\[
\Omega_j,\qquad j=0,1,\ldots,N,
\]
такие, что
\[
\overline\Omega\subset\bigcup_{j=1}^{N}\Omega_j,
\qquad
\overline{Q}\subset\bigcup_{j=0}^{N}\Omega_j
\qquad
\overline{\Omega_j}\subset O_j,
\quad j=1,\ldots,N,
\]
и
\[
\overline{\Omega_0}\cap\overline\Omega=\varnothing.
\]
\end{mylemma}

\begin{myproof}
\begin{figure}[H]
        \centering
        \includegraphics[width=0.7\linewidth]{pics/11.png}
        \caption{}
\end{figure}

    Заведём $K_1$
\[
K_1:=\overline \Omega\setminus\bigcup_{j=2}^{N}O_j
\]
Множество $K_1$ замкнуто и $K_1\subset O_1$. Тогда существует $\Omega_1: \quad K_1\subset\Omega_1,\,\, \overline{\Omega_1}\subset O_1$. \\
Это множество можно вписать туда через расстояние как в предыдущей лемме.
При этом сохраняется то, что $\overline{\Omega}\subset\Omega_1\cup\bigcup_{j=2}^{N}O_j$. \\
Теперь заводим $K_2:=\overline \Omega\setminus\Omega_1\setminus\bigcup_{j=3}^{N}O_j$. \\
\[K_2 \subset O_2, \qquad K_2 - \text{замкнутое}\].
Теперь существует такое $\Omega_2$ - открытое, что 
\[K_2\subset \Omega_2, \qquad \overline{\Omega_2}\subset O_2; \qquad \overline{\Omega}\subset\Omega_1\cup\Omega_2\cup\bigcup_{j=3}^{N}O_j\]
И так далее. Таким образом мы получим набор $\Omega_j$ такой, что:
\[\overline{\Omega}\subset\bigcup_{j=1}^{N}\Omega_j\] 
Пусть $\overline{Q}\subset\widetilde{Q}$ - открытое. И у нас есть $K_0:=\overline{Q}\setminus\bigcup_{j=1}^{N}\Omega_j$ - замкнуто и $K_0\subset\widetilde{Q}\setminus\overline{\Omega}$. Следовательно, существует $\Omega_0$ - открытое, такое, что $K_0\subset\Omega_0, \,\, \overline{\Omega_0} \subset \widetilde{Q}\setminus\overline{\Omega} \,\,(\Rightarrow \overline{\Omega_0}\cap\overline{\Omega}=\varnothing)$.
Условию с вложением $\overline{Q}$ мы удовлетворили, точка, которая лежит в кубе лежит либо в объединении $\Omega_j$ начиная с единицы, либо она лежит в $K_0\subset\Omega_0$. 


\end{myproof}

\begin{mytheorem}
Пусть $\Omega$ --- открытая ограниченная область,
\[
\overline\Omega\subset\bigcup_{j=1}^{N}O_j,
\qquad O_j\text{ --- открытые}.
\]
Тогда существуют функции
\[
\{\xi_j\}_{j=1}^{N},
\qquad
\xi_j\in C_0^\infty(\mathbb R^n),
\]
такие, что
\[
\operatorname{supp}\xi_j\subset O_j,
\qquad
0\le\xi_j(x)\le1,
\]
и
\[
\sum_{j=1}^{N}\xi_j(x)=1,
\qquad \forall x\in\overline\Omega.
\]
\end{mytheorem}

\begin{myproof}
Пусть $\overline{\bigcup_{j=1}^{N}O_j}\subset Q$.
По предыдущей лемме выберем $\Omega_j$, $j=0,\ldots,N$, при чём $\overline{\Omega_j}\subset O_j$.
Для $j=1,\ldots,N$ по первой лемме существуют функции
\[
\eta_j\in C_0^\infty(\mathbb R^n),
\qquad
\eta_j(x)\ge0,
\]
такие, что
\[
\eta_j\big|_{\overline{\Omega_j}}=1,
\qquad
\operatorname{supp}\eta_j\subset O_j.
\]
Для $j=0$ аналогично выбираем $\overline{\Omega_0}\subset\widetilde{Q}\setminus\overline{\Omega}$
\[
\eta_0\in C_0^\infty(\mathbb R^n),
\qquad
\eta_0(x)\ge0,
\qquad
\eta_0\big|_{\overline{\Omega_0}}=1,
\qquad
\operatorname{supp}\eta_0\subset\widetilde Q\setminus\overline\Omega.
\]

Определим, $j=1,\ldots,N$,
\[
\xi_j(x)=
\begin{cases}
\displaystyle
\frac{\eta_j(x)}{\sum\limits_{i=0}^{N}\eta_i(x)}, & x\in Q,\\[4mm]
0, & x\notin Q.
\end{cases}
\]
На $\overline Q$ (хотя бы одна из функций равна 1)
\[
\sum_{i=0}^{N}\eta_i(x)\ge1.
\]
А на $x\notin Q$ нужно продолжить функцию нулём.
Поэтому
\[
\xi_j\in C_0^\infty(\mathbb R^n),
\qquad
\operatorname{supp}\xi_j\subset O_j.
\]
Так как при $x\in\overline\Omega$ имеем $\eta_0(x)=0$ (носитель не пересекается с $\overline{\Omega}$), то
\[
\sum_{j=1}^{N}\xi_j(x)=1,
\qquad x\in\overline\Omega.
\]
\end{myproof}

\section{Интегральные функционалы}

\subsection{Функционалы в нормированных пространствах}

Пусть $X$ --- линейное нормированное пространство и функционал
\[
J:X\to\mathbb R.
\]

\begin{mydef}
Вариацией функционала $J$ в точке $f$ в направлении $h$ называется
\[
\delta J(f,h)
=
\left.\frac{\dd}{\dd\alpha}J(f+\alpha h)\right|_{\alpha=0}
\]
Определение корректно в случае существования производной (что вообще говоря не обязательно)
\end{mydef}

\begin{myremark}
\begin{enumerate}
\item Производная по Гато:
\[
\delta J(f,h)=J'_G[f]h.
\]
\item Вариация однородна по $h$.

\[
\delta J(f,\lambda h)
=
\lim_{\alpha\to 0}
\frac{
J(f+\alpha\lambda h)-J(f)
}{
\alpha
}=\begin{vmatrix}
    \beta=\alpha\lambda
\end{vmatrix}
=
\lim_{\beta\to 0}
\frac{
J(f+\beta h)-J(f)
}{
\beta/\lambda
}
=\]\[=
\lambda
\lim_{\beta\to 0}
\frac{
J(f+\beta h)-J(f)
}{
\beta
}
=
\lambda \delta J(f,h).
\]

Таким образом,
\[
\delta J(f,\lambda h)
=
\lambda\,\delta J(f,h).
\]
\end{enumerate}
\end{myremark}

\begin{myexe}
Линейности по $h$ в общем случае нет.
\end{myexe}

Пусть задан линейный функционал
\[
\varphi:X\to\mathbb R
\]

\begin{mydef}
Функционал $\varphi$ называется ограниченным, если существует $C>0$ такое, что
\[
|\varphi(x)|\le C\|x\|_X,
\qquad \forall x\in X.
\]
\end{mydef}

\begin{myexample}
Функционал не обязан быть ограниченным. Пусть
\[
X=C[-1,1] - \text{непр. ф-ии},
\qquad
\|f\|_1=\int_{-1}^{1}|f(x)|\,\dd x,
\]
и
\[
\varphi(f)=f(0).
\]
Рассмотрим функции
\[
f_n(x)=
\begin{cases}
1-n|x|, & |x|\le \dfrac1n,\\[2mm]
0, & |x|>\dfrac1n.
\end{cases}
\]
Тогда
\[
\varphi(f_n)=1,
\qquad
\|f_n\|_1=\frac1n - \text{площадь треугольника}.
\]

\begin{center}

\draw[->] (-2.2,0)--(2.2,0);

\node[left] at (0,1) {$1$};
\end{tikzpicture}
\end{center}
\end{myexample}

\begin{myexe}
Если $\dim X<\infty$, то всякий линейный функционал на $X$ ограничен.
\end{myexe}

\begin{mydef}
Сопряжённым пространством к $X$ - линейному называется
\[
X'=X^*=
\{\text{линейные ограниченные функционалы на }X\}.
\]
Норма задаётся формулой
\[
\|\varphi\|_{X'}
=
\sup_{f\ne0}\frac{|\varphi(f)|}{\|f\|_X}.
\]
\end{mydef}

\begin{myexe}
\begin{enumerate}
\item Это действительно норма.
\item $X'$ --- банахово(полное) пространство.
\end{enumerate}
\end{myexe}



\begin{mydef}
    Пусть $J:X\to\mathbb R$ --- функционал и $f\in X$.
Говорят, что $J$ дифференцируем по Фреше в точке $f$, если существует $\varphi\in X'$ такое, что
\[
J[f+h]
=
J[f]+\varphi(h)+o(\|h\|_X),
\qquad h\to0.
\]
При этом
\[
J'_F[f]h=\varphi(h).
\]
\end{mydef}

\begin{myexe}
\begin{enumerate}
\item Если $\varphi$ существует, то он единственный.
\item Если существует $J'_F[f]$, то существует производная по Гато
\[
J'_G[f]h=J'_F[f]h
\qquad \forall h.
\]
\item Существование $J'_G[f]h$ для всех $h$ не влечёт существования $J'_F[f]$.
\end{enumerate}
\end{myexe}

\begin{mydef}
Говорят, что
\[
J\in C^1(X),
\]
если для каждого $f\in X$ существует $J'_F[f]\in X'$ и отображение
\[
f\longmapsto J'_F[f]
\]
непрерывно как отображение $X\to X'$.
\end{mydef}

Пусть $X$ --- нормированное пространство и
\[
J:X\to\mathbb R.
\]

\begin{mydef}
Точка $f$ называется точкой локального минимума $J$, если
\[
\exists\delta>0:\qquad
J[f]\le J[g]
\quad\forall g\in X:\ \|g-f\|_X<\delta.
\]
Точка $f$ называется точкой строгого локального минимума, если
\[
\exists\delta>0:\qquad
J[f]<J[g]
\quad\forall g\ne f:\ \|g-f\|_X<\delta.
\]
Точка $f$ называется точкой глобального (абсолютного) минимума, если
\[
J[f]\le J[g]
\qquad\forall g\in X.
\]
\end{mydef}

\begin{mylemma}
Пусть $f$ --- локальный экстремум $J$ и вариация $\delta J(f,h)$ существует.
Тогда
\[
\delta J(f,h)=0
\qquad\forall h\in X.
\]
\end{mylemma}

\begin{myproof}
Положим
\[
\psi(\alpha)=J(f+\alpha h).
\]
Из условия $f$ - локальный экстремум $J$: точка $\alpha=0$ является локальным экстремумом $\psi$. Если $\psi'(0)$ существует, то
\[
\psi'(0)=0.
\]
Следовательно,
\[
\delta J(f,h)=0.
\]
\end{myproof}

\begin{mydef}
Точка $f$ называется стационарной точкой функционала $J$, если
\[
\delta J(f,h)=0
\qquad\forall h.
\]
\end{mydef}

\subsection{Пример пространства $C^1$ и непрерывность интегрального функционала}

Пусть
\[
X=C^1([a,b],\mathbb R^n),
\qquad
\|f\|_X=
\max_{[a,b]}|f'(x)|+
\max_{[a,b]}|f(x)|.
\]

\begin{myexe}
\begin{enumerate}
\item $\|\cdot\|_X$ является нормой.
\item $X$ --- банахово пространство.
\end{enumerate}
\end{myexe}

\begin{mytheorem}
Пусть
\[
L\in C([a,b]\times\mathbb R^n\times\mathbb R^n),
\]
и
\[
J[y]=\int_a^bL\bigl(x,y(x),y'(x)\bigr)\,\dd x.
\]
Тогда $J:X\to\mathbb R$ непрерывен.
\end{mytheorem}

\begin{myproof}
Зафиксируем $y_0\in X$, т.к. она непрерывна на $[a,b]$, то она ограничена и её производная аналогично:
\[
|y_0(x)|\le R_0,
\qquad
|y_0'(x)|\le R_1,
\qquad x\in[a,b].
\]
Сузим функцию $L$ так, чтобы она была непрерывна на компакте (+1 нужен чтобы функции $y, y'$ попали в шар)
\[
[a,b]\times\overline B_{R_0+1}\times\overline B_{R_1+1},
\]
поэтому равномерно непрерывна на нём, что означает, что для каждого $\varepsilon>0$ существует $\delta\in(0,1)$ такое, что из
\[
|x-x_0|<\delta,
\qquad
|y-y_0|<\delta,
\qquad
|v-v_0|<\delta
\]
следует
\lbpa{zero}{
\[
|L(x,y,v)-L(x_0,y_0,v_0)|<\frac{\varepsilon}{b-a}.
\]}

Пусть есть такая функция $y$, что
\[
\|y-y_0\|_X<\delta,
\]
то
\[
|y(x)-y_0(x)|<\delta,
\qquad
|y'(x)-y_0'(x)|<\delta,
\]
и потому из \ref{zero} следует
\[
\left|L(x,y(x),y'(x))-L(x,y_0(x),y_0'(x))\right|
<\frac{\varepsilon}{b-a}.
\]
Следовательно,
\[
|J[y]-J[y_0]|
\le
\int_a^b
\left|L(x,y,y')-L(x,y_0,y_0')\right|\,\dd x
<\varepsilon.
\]
\end{myproof}

\begin{mytheorem}
Пусть
\[
L\in C^1([a,b]\times\mathbb R^n\times\mathbb R^n),
\]
и
\[
J[y]=\int_a^bL\bigl(x,y(x),y'(x)\bigr)\,\dd x.
\]
Тогда (в любой точке есть производная по Фреше)
\[
J\in C^1(X),
\]
\[J(y+h)=J(y)+\delta J(y; h)+o(\|h\|_X)\]
а вариация имеет вид
\[
\boxed{
\delta J(y;h)
=
\int_a^b
\left(
\left\langle\nabla_yL(x,y,y'),h(x)\right\rangle
+
\left\langle\nabla_vL(x,y,y'),h'(x)\right\rangle
\right)\dd x.}
\]
\end{mytheorem}

\begin{myproof}
Используем разложение Тейлора, поскольку она из $C^1$
\[
L(x,y+h,v+w)
=
L(x,y,v)
+
\langle\nabla_yL(x,y,v),h\rangle
+
\langle\nabla_vL(x,y,v),w\rangle
+
o(|h|+|w|),
\]
при $h,w\to0$ в $\mathbb R^n$.
Подставляя
\[
y=y(x),\qquad h=h(x),\qquad v=y'(x),\qquad w=h'(x)
\]
и интегрируя по $[a,b]$, получаем
\[
J[y+h]
=
J[y]
+\underbrace{
\int_a^b
\left(
\langle\nabla_yL(x,y,y'),h\rangle
+
\langle\nabla_vL(x,y,y'),h'\rangle
\right)\dd x}_{\text{линейный ограниченный функционал по h}}
+
o(\|h\|_X).
\]
Про интегрирование о-малого остаётся упражнение:)
\end{myproof}

\begin{myexe}
Отображение
\[
y\longmapsto J'_F[y],
\qquad X\to X',
\]
непрерывно.
\end{myexe}

\begin{myremark}
При $n=1$
\[
\delta J(y;h)
=
\int_a^b
\left(
\frac{\partial L(x,y,y')}{\partial y}\,h(x)
+
\frac{\partial L(x,y,y')}{\partial v}\,h'(x)
\right)\dd x.
\]
\end{myremark}

\subsection{Леммы}

\begin{mylemma} Лемма Лагранжа

Пусть $\Omega\subset\mathbb R^n$ --- открытое множество и
\[
f\in C(\Omega).
\]
Если
\[
\int_\Omega f(x)h(x)\,\dd x=0
\qquad
\forall h\in C_0^\infty(\Omega) (\supp h \subset \Omega - \text{компакт}),
\]
то
\[
f\equiv0.
\]
\end{mylemma}

\begin{myproof}
Предположим противное. Тогда существует $a\in\Omega$ такое, что, например,
\[
f(a)>0.
\]
Тогда существует $\delta>0$ такое, что
\[\overline{B_\delta(a)}\subset\Omega,
\qquad
f(x)>0\quad\forall x\in B_\delta(a).
\]
Выберем
\[
h(x)=
\begin{cases}
\displaystyle
\exp\!\left(-\frac{1}{\delta^2-|x-a|^2}\right), & |x-a|<\delta,\\[2mm]
0, & |x-a|\ge\delta.
\end{cases}
\]
Тогда $h\in C_0^\infty(\Omega)$ и
\[
\int_\Omega f h\,\dd x
=
\int_{B_\delta(a)}fh\,\dd x>0,
\]
противоречие.
\end{myproof}

\begin{mylemma} Лемма Дюбуа-Реймона

Пусть
\[
g\in C([a,b],\mathbb R^n).
\]
Тогда
\[
\int_a^b\langle g(x),h'(x)\rangle\,\dd x=0
\]
для всех
\[
h\in C^1([a,b],\mathbb R^n),
\qquad
h(a)=h(b)=0,
\]
тогда и только тогда, когда
\[
g(x)=\mathrm{const}.
\]
\end{mylemma}

\begin{myproof}
Обратное утверждение очевидно.
Для прямого положим
\[
c:=\frac1{b-a}\int_a^b g(x)\,\dd x
\]
и
\[
h(x)=\int_a^x(g(t)-c)\,\dd t.
\]
Тогда
\[
h'(x)=g(x)-c,
\qquad
h(a)=h(b)=0.
\]
Следовательно,
\[
0
=
\underbrace{\int_a^b\langle g(x),\underbrace{h'(x)}_{=g(x)-c}\rangle\,\dd x}_{=0}
-
\underbrace{\int_a^b\langle c,\underbrace{h'(x)}_{=g(x)-c}\rangle\,\dd x}_{=0}
=
\int_a^b|g(x)-c|^2\,\dd x.
\]
Отсюда
\[
g(x)=c.
\]
\end{myproof}

\begin{mylemma} Обобщённая лемма Дюбуа-Реймона

Пусть
\[
g\in C[a,b].
\]
Тогда
\[
\int_a^b g(x)h''(x)\,\dd x=0
\]
для всех $h\in C^2[a,b]$, удовлетворяющих
\[
h(a)=h(b)=h'(a)=h'(b)=0,
\]
тогда и только тогда, когда
\[
\boxed{g(x)=Ax+B.}
\]
\end{mylemma}

\begin{myproof}
Положим
\[
h(x)
=
\int_a^x\int_a^t
\bigl(g(\tau)-\alpha(\tau-a)-\beta\bigr)
\,\dd\tau\,\dd t,
\qquad
\alpha,\beta\in\mathbb R.
\]
Тогда
\[
h(a)=0,
\qquad
h'(x)=\int_a^x\bigl(g(\tau)-\alpha(\tau-a)-\beta\bigr)\,\dd\tau,
\qquad
h'(a)=0,
\]
и
\[
h''(x)=g(x)-\alpha(x-a)-\beta.
\]
Условия $h'(b)=0$ и $h(b)=0$ дают систему относительно $\alpha,\beta$:
\[
\int_a^b g(\tau)\,\dd\tau
-
\alpha\frac{(b-a)^2}{2}
-
\beta(b-a)=0,
\]
\[
\int_a^b\int_a^t g(\tau)\,\dd\tau\,\dd t
-
\alpha\frac{(b-a)^3}{6}
-
\beta\frac{(b-a)^2}{2}=0.
\]
Хотим понять существуют ли такие $\alpha, \beta$, поэтому берём их за переменные и если определитель системы не равен 0, то для любой правой части будет единсвтенное решение. Её определитель равен
\[
\begin{vmatrix}
\dfrac{(b-a)^2}{2} & b-a\\[2mm]
\dfrac{(b-a)^3}{6} & \dfrac{(b-a)^2}{2}
\end{vmatrix}
=
\frac{(b-a)^4}{12}>0,
\]
поэтому $\alpha,\beta$ существуют.

Тогда
\[
0
=
\int_a^b g(x)h''(x)\,\dd x
-
\underbrace{\int_a^b\bigl(\alpha(x-a)+\beta\bigr)h''(x)\,\dd x}_{\text{можем вписать любую линейную функцию}}
\]
\[
\int_a^b
\bigl(g(x)-\alpha(x-a)-\beta\bigr)h''(x)\,\dd x=0.
\]
Но
\[
h''(x)=g(x)-\alpha(x-a)-\beta,
\]
следовательно,
\[
g(x)=\alpha(x-a)+\beta=Ax+B.
\]
\end{myproof}

\begin{myexe}
Пусть $g\in C[a,b]$, $k\in\mathbb N$ и
\[
\int_a^b g(x)h^{(k)}(x)\,\dd x=0
\]
для всех $h\in C^k[a,b]$, удовлетворяющих
\[
h(a)=h'(a)=\cdots=h^{(k-1)}(a)=0,
\]
\[
h(b)=h'(b)=\cdots=h^{(k-1)}(b)=0.
\]
Тогда и только тогда $g$ --- многочлен степени не выше $k-1$.
\end{myexe}

\subsection{Задача с фиксированными граничными точками}

\begin{mytheorem}
Пусть
\[
L\in C^1([a,b]\times\mathbb R^n\times\mathbb R^n),
\]
\[
J[y]=\int_a^bL(x,y,y')\,\dd x,
\]
и $y_0$ --- локальный экстремум $J$ на множестве
\[
\left\{y\in C^1([a,b],\mathbb R^n):\ y(a)=A,\ y(b)=B\right\}.
\]
Тогда существует
\[
\boxed{
\frac{\dd}{\dd x}
\nabla_vL\bigl(x,y_0(x),y_0'(x)\bigr)
=
\nabla_yL\bigl(x,y_0(x),y_0'(x)\bigr).}
\]
\end{mytheorem}

\begin{myproof}
Пусть
\[
h\in C^1([a,b],\mathbb R^n),
\qquad h(a)=h(b)=0.
\]
Тогда $\alpha=0$ --- локальный экстремум функции $J(y_0+\alpha h)$, поэтому
\[
\delta J(y_0;h)=0.
\]
Следовательно,
\[
\int_a^b
\left(
\langle\nabla_yL(x,y_0,y_0'),h\rangle
+
\langle\nabla_vL(x,y_0,y_0'),h'\rangle
\right)\dd x=0.
\]
Мы могли бы проинтегрировать по частям 2-е слагаемое, но мы не знаем дифференцируем ли градиент, поэтому лучше начнём с 1-ого слагаемого.
Положим
\[
G(x):=\int_a^x\nabla_yL\bigl(t,y_0(t),y_0'(t)\bigr)\,\dd t.
\]
Тогда
\[
G\in C^1([a,b],\mathbb R^n)
\]
и
\[
\int_a^b\langle\underbrace{\nabla_yL}_{\text{=G'(x)}},h\rangle\,\dd x
=
\langle G(x),h(x)\rangle\Big|_a^b
-
\int_a^b\langle G(x),h'(x)\rangle\,\dd x.
\]
Граничный член равен нулю, поэтому
\[
\int_a^b
\left\langle
\nabla_vL(x,y_0,y_0')-G(x),h'(x)
\right\rangle\dd x=0.
\]
По лемме Дюбуа--Реймона
\[
\nabla_vL(x,y_0,y_0')=G(x)+\mathrm{const}.
\]
В правой части функция из $C^1$ (т.е. существует производная), следовательно,
\[
\frac{\dd}{\dd x}\nabla_vL(x,y_0,y_0')
=
G'(x)
=
\nabla_yL(x,y_0,y_0').
\]
\end{myproof}

\begin{myremark}
Из доказательства не следует, что $y_0\in C^2$
\end{myremark}

\begin{myexample}
Пусть $n=1$, $[a,b]=[-1,1]$ и
\[

\begin{mytheorem}
Пусть выполнены условия предыдущей теоремы,
\[
L\in C^2([a,b]\times\mathbb R^n\times\mathbb R^n),
\]
и
\[
\det\left(
\frac{\partial^2L(x,y,v)}{\partial v_j\partial v_k}
\right)_{j,k=1}^{n}
\ne0
\qquad \forall x,y,v.
\]
Тогда
\[
y_0\in C^2([a,b],\mathbb R^n).
\]
\end{mytheorem}

\begin{myproof}
Рассмотрим отображение
\[
\Phi:(x,y,v)\longmapsto
\bigl(x,y,\nabla_vL(x,y,v)\bigr).
\]
Имеем
\[
\Phi:[a,b]\times\mathbb R^{2n}	\longrightarrow [a,b]\times\mathbb R^{2n},
\qquad
\Phi\in C^1.
\]
Матрица $D\Phi(x,y,v)$ имеет блочно-треугольный вид:
\[D\Phi=
\left(
\begin{array}{cccc|ccc@{}l}
1 & 0 & \cdots & 0 & 0 & \cdots & 0
  & \rlap{\hspace{2em}$\leftarrow$ \text{$x$ по всем переменным}} \\
0 & 1 & \cdots & 0 & 0 & \cdots & 0
  & \rlap{\hspace{2em}$\leftarrow$ \text{$y_1$ по всем переменным}} \\
\vdots & \vdots & \ddots & \vdots & \vdots & & \vdots
  & \rlap{\hspace{2.3em}$\vdots$} \\
0 & 0 & \cdots & 1 & 0 & \cdots & 0
  & \rlap{\hspace{2em}$\leftarrow$ \text{$y_n$ по всем переменным}} \\
\hline
\dfrac{\partial^2 L}{\partial x\,\partial v_1} &
\dfrac{\partial^2 L}{\partial y_1\,\partial v_1} &
\cdots &
\dfrac{\partial^2 L}{\partial y_n\,\partial v_1} &
\dfrac{\partial^2 L}{\partial v_1^2} &
\cdots &
\dfrac{\partial^2 L}{\partial v_1\,\partial v_n}
  & \rlap{\hspace{2em}$\leftarrow$ \text{$\dfrac{\partial L}{\partial v_1}$ по всем переменным}} \\
\vdots & \vdots & \ddots & \vdots & \vdots & \ddots & \vdots
  & \rlap{\hspace{2.3em}$\vdots$} \\
\dfrac{\partial^2 L}{\partial x\,\partial v_n} &
\dfrac{\partial^2 L}{\partial y_1\,\partial v_n} &
\cdots &
\dfrac{\partial^2 L}{\partial y_n\,\partial v_n} &
\dfrac{\partial^2 L}{\partial v_n\,\partial v_1} &
\cdots &
\dfrac{\partial^2 L}{\partial v_n^2}
  & \rlap{\hspace{2em}$\leftarrow$ \text{$\dfrac{\partial L}{\partial v_n}$ по всем переменным}}
\end{array}
\right)
\hphantom{\hspace{2em}\leftarrow\ \text{$\dfrac{\partial L}{\partial v_n}$ по всем переменным}}
\]
\[
\det D\Phi
=
\det\left(
\frac{\partial^2L}{\partial v_j\partial v_k}
\right)
\neq0.
\]

Пусть $x_0\in(a,b)$, рассмотрим точку $(x_0, y_0(x_0),y_0'(x_0))$,  по теореме об обратной функции в окрестности существует обратное отображение $\psi\in C^1$ такое, что

\begin{mytheorem}
Пусть
\[
L\in C^1([a,b]\times\mathbb R^n\times\mathbb R^n),
\]
и $y_0$ --- локальный экстремум функционала
\[
J[y]=\int_a^bL(x,y,y')\,\dd x
\]
на всём пространстве $X=C^1([a,b],\mathbb R^n)$.
Тогда
\lbpa{2}{\[
\boxed{
\frac{\dd}{\dd x}\nabla_vL(x,y_0,y_0')
=
\nabla_yL(x,y_0,y_0')}
\]}
и
\lbpa{3}{\[
\boxed{
\nabla_vL(a,y_0(a),y_0'(a))
=
\nabla_vL(b,y_0(b),y_0'(b))
=0.}
\]}
\end{mytheorem}

\begin{myproof}
Для любого $h\in X$
\[
\delta J(y_0,h)=0,
\]
то есть
\lbpa{1}{\[
\int_a^b
\left(
\langle\nabla_yL(x, y_0, y_0'),h(x)\rangle
+
\langle\nabla_vL(x, y_0, y_0'),h'(x)\rangle
\right)\dd x=0 \qquad \forall h
\]}
В частности выполняется для $h$ с 0 на концах. Т.е. из предыдущего параграфа существует $\frac{\dd}{\dd x}\nabla_vL(x,y_0,y_0')
=
\nabla_yL(x,y_0,y_0')$.
Теперь можем проинтегрировать \ref{1}, получим 
\[\int_a^b
\langle\nabla_yL(x, y_0, y_0')-\frac{\dd}{\dd x}(\nabla_vL(x,y_0,y_0')),h(x)\rangle dx
+\langle\nabla_vL(x,y_0,y_0'), h_x\rangle\Big|_a^b=0 \qquad \forall h\in X
\]
Первое слагаемое равно 0, тогда т.к. для любых $h$, то понятно, что на концах равным 0 может быть только градиент в обоих точках(можем взять такую $h: \quad h(b)=0 \, h(a)=e_j$):
\[
\nabla_vL(a,y_0(a),y_0'(a))
=
\nabla_vL(b,y_0(b),y_0'(b))
=0.
\]
\end{myproof}

\end{myremark}

\subsection{Частные случаи уравнения Эйлера--Лагранжа}

\begin{enumerate}
\item Если

\]

Это не дифференциальное уравнение.

\]
В соответствующем случае, если зависит только от $(x,v)$:
\[
\nabla_vL(x,y_0,y_0')=\vec c,
\]
откуда
\[
y_0'(x)=f(x,\vec c),
\qquad
 y_0(x)=\int_a^x f(t,\vec c)\,\dd t+\vec c_2.
\]

\item Если
\[
L=L(y,v),
\]
то
\[\frac{\dd}{\dd x}
\left(

\langle y_0',\nabla_vL(y_0,y_0')\rangle
\right)=\]

\end{enumerate}

Пусть

\qquad \forall f\in M.
\]
Тогда $M$ --- гиперповерхность.

Касательное пространство к $M$ в точке $u\in M$ имеет вид
\[

\]

В случае $X=\mathbb R^n$
\[
G[u+h]
=
G[u]+\langle\nabla G(u),h\rangle+o(|h|),
\qquad h\to0.
\]

\begin{mylemma}
Пусть $J,G\in C^1(X)$,
\[
M=\{f\in X:\ G[f]=0\}.
\]
Пусть $u$ --- локальный экстремум $J$ на $M$, причём $u$ не является стационарной точкой $G$.
Если
\[
\delta G(u;h)=0,
\]
то
\[
\delta J(u;h)=0.
\]
\end{mylemma}

\begin{myproof}
По предыдущей теореме существует кривая
\[
w(t)=u+s(t)v+th,
\qquad
w(t)\in M,
\qquad
w(0)=u,
\qquad
w'(0)=h.
\]
Поскольку $u$ --- локальный экстремум $J$ на $M$, точка $t=0$ является локальным экстремумом функции $J[w(t)]$, поэтому
\[
\left.\frac{\dd}{\dd t}J[w(t)]\right|_{t=0}=0.
\]
Разложение
\[
J[w(t)]
=
J[u]
+s(t)\delta J(u;v)
+t\delta J(u;h)
+o(\|s(t)v+th\|_X)
\]
и $s(t)=o(t)$ дают
\[
\delta J(u;h)=0.
\]
\end{myproof}

\begin{mytheorem}
Пусть $J,G\in C^1(X)$,
\[
M=\{f:\ G[f]=0\},
\]
и $u$ --- локальный экстремум $J$ на $M$, причём $u$ не является стационарной точкой $G$.
Тогда существует $\lambda\in\mathbb R$ такое, что $u$ является стационарной точкой функционала
\[
J+\lambda G.
\]
\end{mytheorem}

\begin{myproof}
Выберем $v\in X$ так, что
\[
\delta G(u;v)\ne0,
\]
и положим
\[
\lambda:=-\frac{\delta J(u;v)}{\delta G(u;v)}.
\]
Для произвольного $y\in X$ положим
\[
h:=y-\frac{\delta G(u;y)}{\delta G(u;v)}v.
\]
Тогда
\[
\delta G(u;h)=0.
\]
По предыдущей лемме
\[
\delta J(u;h)=0.
\]
Следовательно,
\[
\delta J(u;y)
-
\frac{\delta G(u;y)}{\delta G(u;v)}\delta J(u;v)=0,
\]
то есть
\[
\delta(J+\lambda G)(u;y)=0
\qquad\forall y\in X.
\]
\end{myproof}

\begin{mytheorem}
Пусть
\[
J,G_1,G_2,\ldots,G_m\in C^1(X),
\]
\[
M=\{f\in X:\ G_1[f]=\cdots=G_m[f]=0\},
\]
и $u$ --- локальный экстремум $J$ на $M$.
Рассмотрим линейный оператор
\[
\vec G'_F[u]:X\to\mathbb R^m,
\qquad
h\longmapsto
\begin{pmatrix}
\delta G_1(u;h)\\
\vdots\\
\delta G_m(u;h)
\end{pmatrix}.
\]
Если
\[
\operatorname{Im}\vec G'_F[u]=\mathbb R^m,
\]
то существуют
\[
\lambda_1,\ldots,\lambda_m\in\mathbb R
\]
такие, что $u$ является стационарной точкой функционала
\[
J+\lambda_1G_1+\cdots+\lambda_mG_m.
\]
\end{mytheorem}

\begin{myremark}
Числа $\lambda_1,\ldots,\lambda_m$ называются множителями Лагранжа.
\end{myremark}

По теореме о множителе Лагранжа существует $\lambda\in\mathbb R$ такое, что
\[
\delta(J+\lambda G)(\rho;h)=0
\qquad\forall h.
\]
Имеем
\[
\left.
\frac{\dd}{\dd\alpha}
\left[
\iint_{\Gamma\times\Gamma}
\frac{(\rho(x)+\alpha h(x))(\rho(y)+\alpha h(y))}{|x-y|}
\,\dd S(x)\dd S(y)
+
\lambda\left(\int_\Gamma(\rho+\alpha h)\,\dd S-q\right)
\right]
\right|_{\alpha=0}
=0.
\]
Отсюда
\[
\iint_{\Gamma\times\Gamma}
\frac{\rho(x)h(y)+\rho(y)h(x)}{|x-y|}
\,\dd S(x)\dd S(y)
+
\lambda\int_\Gamma h(x)\,\dd S(x)=0,
\]
то есть
\[
\int_\Gamma
\left(
2\int_\Gamma\frac{\rho(y)}{|x-y|}\,\dd S(y)+\lambda
\right)
 h(x)\,\dd S(x)=0.
\]
Следовательно,
\[
\boxed{
\int_\Gamma\frac{\rho(y)}{|x-y|}\,\dd S(y)
=-\frac{\lambda}{2},
\qquad x\in\Gamma.}
\]
Это задача Робена.
Если
\[
\varphi(x)=\int_\Gamma\frac{\rho(y)}{|x-y|}\,\dd S(y),
\]
то
\[
\Delta\varphi(x)=0\quad\text{в }\Omega,
\qquad
\varphi\big|_\Gamma=-\frac\lambda2,
\]
откуда
\[
\varphi(x)=-\frac\lambda2
\qquad\text{в }\Omega.
\]
\end{myexample}

\subsection{Минимизирующая последовательность}

Рассмотрим функционал
\[
J[y]
=
\int_a^b
\left(
 p(x)y'(x)^2+q(x)y(x)^2
\right)\dd x,
\]
где
\[
p\in C^1[a,b],
\qquad
p(x)>0\quad\forall x,
\qquad
q\in C[a,b].
\]
Пусть
\[
M_1=
\left\{
 y\in C^1[a,b]:
 y(a)=y(b)=0,
 \int_a^b y(x)^2\,\dd x=1
\right\}.
\]

\begin{myremark}
Если $y_0$ --- локальный экстремум, то по теореме о множителе Лагранжа существует $\lambda\in\mathbb R$ такое, что $y_0$ --- стационарная точка функционала
\[
\int_a^bF\,\dd x,
\qquad
F(x,y,v)=p(x)v^2+q(x)y^2-\lambda y^2.
\]
Поэтому
\[
\frac{\dd}{\dd x}\bigl(2p(x)y_0'(x)\bigr)
=
2(q(x)-\lambda)y_0(x),
\]
то есть
\[
\boxed{
(p y_0')'+(\lambda-q)y_0=0,}
\qquad
\boxed{y_0(a)=y_0(b)=0.}
\]
\end{myremark}

Положим
\[
\lambda_1:=\inf_{M_1}J[y].
\]

\begin{mylemma}
Если
\[
q_0:=\min_{[a,b]}q(x),
\]
то
\[
\lambda_1\ge q_0.
\]
\end{mylemma}

\begin{myproof}
Для $y\in M_1$
\[
J[y]
\ge
\int_a^bq(x)y(x)^2\,\dd x
\ge
q_0\int_a^b y(x)^2\,\dd x
=q_0.
\]
\end{myproof}

\begin{mylemma}
Если
\[
f\in C^1[a,b],
\qquad
f(a)=f(b)=0,
\]
то
\[
\boxed{
J[f]\ge\lambda_1\int_a^b f(x)^2\,\dd x.}
\]
\end{mylemma}

\begin{myproof}
Положим
\[
y(x)=
\frac{f(x)}{
\sqrt{\displaystyle\int_a^b f(x)^2\,\dd x}}.
\]
Тогда $y\in M_1$, следовательно
\[
J[y]\ge\lambda_1.
\]
После умножения на $\int_a^b f^2\,\dd x$ получаем требуемое.
\end{myproof}

Существует минимизирующая последовательность
\[
\{f_m\}\subset M_1,
\qquad
J[f_m]\to\lambda_1.
\]

\begin{mytheorem}
\textbf{Теорема Арцела--Асколи.}\par
Пусть
\[
\{h_k\}_{k=1}^{\infty}\subset C[a,b]
\]
--- равномерно ограниченное и равностепенно непрерывное семейство. Тогда существует подпоследовательность
\[
\{h_{k_j}\}_{j=1}^{\infty}
\]
и функция $h_*\in C[a,b]$ такие, что
\[
h_{k_j}\to h_*
\]
равномерно на $[a,b]$.
\end{mytheorem}

\begin{mylemma}
Пусть
\[
f_m\in C^1[a,b],
\qquad
f_m(a)=0,
\]
и существует $K>0$ такое, что
\[
\int_a^b f_m'(x)^2\,\dd x\le K.
\]
Тогда существует подпоследовательность $\{f_{m_j}\}$, сходящаяся равномерно на $[a,b]$ к некоторой функции $y\in C[a,b]$.
\end{mylemma}

\begin{myproof}
Пусть $x_1<x_2$. Тогда
\[
|f_m(x_2)-f_m(x_1)|
=
\left|\int_{x_1}^{x_2}f_m'(t)\,\dd t\right|
\le
\sqrt{x_2-x_1}
\sqrt{\int_{x_1}^{x_2}|f_m'(t)|^2\,\dd t}
\le
\sqrt{x_2-x_1}\,\sqrt K.
\]
Следовательно, семейство $\{f_m\}$ равностепенно непрерывно.
Кроме того, при $x_1=a$, $x_2=x$ получаем
\[
|f_m(x)|\le\sqrt{(b-a)K},
\]
то есть семейство равномерно ограничено. По теореме Арцела--Асколи существует равномерно сходящаяся подпоследовательность.
\end{myproof}

\begin{mylemma}
Пусть
\[
p_0:=\min_{[a,b]}p(x)>0.
\]
Тогда для $f\in M_1$
\[
\boxed{
\int_a^b f'(x)^2\,\dd x
\le
\frac{J[f]-q_0}{p_0}.}
\]
\end{mylemma}

\begin{myproof}
Имеем
\[
J[f]
\ge
p_0\int_a^b f'(x)^2\,\dd x
+
q_0\int_a^b f(x)^2\,\dd x.
\]
Так как $f\in M_1$, последний интеграл равен $1$.
\end{myproof}

\subsection{Свойства предельной функции}

Введём оператор

\left(p(x)\frac{\dd}{\dd x}\right)

\{f\in C^2[a,b]:\ f(a)=f(b)=0\}.

\begin{mylemma}
Для любого $h\in\operatorname{Dom}\mathcal L$
\[
\boxed{
\int_a^b y_1\,\mathcal L h\,\dd x
=
\lambda_1\int_a^b y_1h\,\dd x.}

\end{mylemma}

\begin{myproof}
Пусть $h\in\operatorname{Dom}\mathcal L$ и $\tau\in\mathbb R$. По предыдущей лемме
\[
J[f_m+\tau h]
\ge
\lambda_1
\int_a^b(f_m+\tau h)^2\,\dd x.

Раскрывая обе части,
\[
J[f_m]
+2\tau\int_a^b(pf_m'h'+qf_mh)\,\dd x
+\tau^2J[h]
\]
\[
\ge
\lambda_1
\int_a^b f_m^2\,\dd x
+2\tau\lambda_1\int_a^b f_mh\,\dd x
+\tau^2\lambda_1\int_a^b h^2\,\dd x.

Переходя к пределу по подпоследовательности и используя

\int_a^b(pf_m'h'+qf_mh)\,\dd x

\int_a^b f_m\,\mathcal Lh\,\dd x,

\lambda_1
+2\tau\int_a^b y_1\mathcal Lh\,\dd x
+\tau^2J[h]
\ge
\lambda_1
+2\tau\lambda_1\int_a^b y_1h\,\dd x
+\tau^2\lambda_1\int_a^b h^2\,\dd x.
\]
Это неравенство имеет вид
\[
A\tau^2+B\tau\ge0
\qquad\forall\tau\in\mathbb R,
\]
откуда
\[
B=0.
\]
Следовательно,
\[
\int_a^b y_1\mathcal Lh\,\dd x
=
\lambda_1\int_a^b y_1h\,\dd x.
\]
\end{myproof}

\begin{mytheorem}
Пусть $y\in C[a,b]$. Тогда

\qquad
\mathcal Ly=\lambda y
\]
тогда и только тогда, когда
\[
\int_a^b y\mathcal Lh\,\dd x
=
\lambda\int_a^b yh\,\dd x
\qquad\forall h\in\operatorname{Dom}\mathcal L.
\]
\end{mytheorem}

\begin{myproof}
Если $y\in\operatorname{Dom}\mathcal L$ и $\mathcal Ly=\lambda y$, то по симметрии
\[
(y,\mathcal Lh)
=
(\mathcal Ly,h)
=
\lambda(y,h).
\]

Обратно, положим
\[
\widetilde q(x)=q(x)-\lambda.
\]
Тогда
\[
\int_a^b y\left[-(ph')'+\widetilde qh\right]\dd x=0
\qquad\forall h\in\operatorname{Dom}\mathcal L.
\]
Выберем функции $g_1,g_2$ так, что
\[
g_1'=yp',
\qquad
 g_1\in C^1,
\]
\[
g_2''=y\widetilde q,
\qquad
 g_2\in C^2.
\]
Тогда после преобразования
\[
\int_a^b(-py+g_1+g_2)h''(x)\,\dd x=0
\]
для всех $h\in C^2[a,b]$ с
\[
h(a)=h'(a)=h(b)=h'(b)=0.
\]
По обобщённой лемме Дюбуа--Реймона
\[
-py+g_1+g_2=Ax+B.
\]
Следовательно,
\[
py=g_1+g_2-Ax-B\in C^1[a,b],
\]
откуда
\[
y\in C^1[a,b].
\]
Дифференцируя,
\[
p'y+py'=g_1'+g_2'-A.
\]
Так как $g_1'=yp'$, получаем
\[
py'=g_2'-A\in C^1[a,b],
\]
и потому
\[
y\in C^2[a,b].
\]
Далее
\[
(py')'=g_2''=\widetilde qy,
\]
то есть
\[
\mathcal Ly=\lambda y.
\]
Наконец, интегрирование по частям в исходном слабом равенстве даёт
\[
-py h'\Big|_a^b=0
\]
для всех допустимых $h$, следовательно
\[
py\Big|_a^b=0.
\]
Так как $p>0$,
\[
y(a)=y(b)=0.
\]
\end{myproof}

\[
J[y_n]=\lambda_n\le\Lambda,
\]
поэтому последовательность $\{y_n\}$ имеет равномерно сходящуюся подпоследовательность
\[
y_{n_k}\to z
\]
на $[a,b]$.
Следовательно,
\[
\int_a^b|y_{n_k}-y_{n_j}|^2\,\dd x\to0
\qquad k,j\to\infty.
\]
Но при $k\ne j$ из ортонормированности
\[
\int_a^b|y_{n_k}-y_{n_j}|^2\,\dd x=2,
\]
противоречие.
\end{myproof}

\begin{myremark}
При $n\to\infty$
\[
\lambda_n=Cn^2+O(1),
\]
где
\[
\boxed{
C=
\left(
\frac{\pi}{\displaystyle\int_a^b\frac{\dd x}{\sqrt{p(x)}}}
\right)^2.}
\]
\end{myremark}

\subsection{Полнота системы собственных функций}

\begin{mytheorem}
Пусть
\[
f\in\operatorname{Dom}\mathcal L,
\qquad
c_k:=\int_a^b f(x)y_k(x)\,\dd x.
\]
Тогда
\[
\sum_{k=1}^{\infty}c_ky_k(x)
\]
сходится к $f$ в $L_2(a,b)$, то есть
\[
\int_a^b
\left|
 f(x)-\sum_{k=1}^{N}c_ky_k(x)
\right|^2\dd x
\longrightarrow0.
\]
\end{mytheorem}

\begin{myproof}
Положим
\[
f_N(x)
=
f(x)-\sum_{k=1}^{N-1}c_ky_k(x).
\]
Тогда
\[
f_N\in\operatorname{Dom}\mathcal L
\]
и
\[
\int_a^b f_N(x)y_k(x)\,\dd x=0,
\qquad k=1,\ldots,N-1.
\]
Следовательно,
\[
J[f_N]
\ge
\lambda_N\int_a^b f_N(x)^2\,\dd x.
\]
С другой стороны,
\[
J[f_N]
=
\int_a^b(\mathcal Lf_N)f_N\,\dd x
\le
\left(\int_a^b(\mathcal Lf_N)^2\,\dd x\right)^{1/2}
\left(\int_a^b f_N^2\,\dd x\right)^{1/2}.
\]
Поэтому
\[
\lambda_N\|f_N\|_{L_2}^2
\le
\|\mathcal Lf_N\|_{L_2}\|f_N\|_{L_2},
\]
то есть
\[
\|f_N\|_{L_2}
\le
\frac{\|\mathcal Lf_N\|_{L_2}}{\lambda_N}.
\]

Имеем
\[
\mathcal Lf_N
=
\mathcal Lf-
\sum_{k=1}^{N-1}c_k\lambda_ky_k.
\]
Для $k=1,\ldots,N-1$
\[
(\mathcal Lf_N,y_k)
=(f_N,\mathcal Ly_k)
=(f_N,\lambda_ky_k)
=0.
\]
Поэтому
\[
\|\mathcal Lf\|_{L_2}^2
=
\|\mathcal Lf_N\|_{L_2}^2
+
\left\|
\sum_{k=1}^{N-1}c_k\lambda_ky_k
\right\|_{L_2}^2,
\]
и
\[
\|\mathcal Lf_N\|_{L_2}
\le
\|\mathcal Lf\|_{L_2}.
\]
Следовательно,
\[
\boxed{
\|f_N\|_{L_2(a,b)}
\le
\frac{\|\mathcal Lf\|_{L_2(a,b)}}{\lambda_N}
\longrightarrow0.}
\]
\end{myproof}

\begin{myprop}
Для $f\in\operatorname{Dom}\mathcal L$
\[
\boxed{
\int_a^b f(x)^2\,\dd x
=
\sum_{k=1}^{\infty}c_k^2.}
\]
\end{myprop}

\begin{myproof}
Так как
\[
\int_a^b
\left(
 f(x)-\sum_{k=1}^{N}c_ky_k(x)
\right)^2\dd x
\to0,
\]
то
\[
\int_a^b f(x)^2\,\dd x
-2\sum_{k=1}^{N}c_k\int_a^b f(x)y_k(x)\,\dd x
+
\int_a^b\left(\sum_{k=1}^{N}c_ky_k(x)\right)^2\dd x
\to0.
\]
Следовательно,
\[
\int_a^b f(x)^2\,\dd x
-\sum_{k=1}^{N}c_k^2
\to0.
\]
\end{myproof}

\begin{myremark}
\begin{enumerate}
\item Для любого $f\in L_2(a,b)$
\[
\int_a^b f(x)^2\,\dd x
=
\sum_{k=1}^{\infty}c_k^2,
\]
так как $\operatorname{Dom}\mathcal L$ плотна в $L_2(a,b)$.
\item Если $f\in\operatorname{Dom}\mathcal L$, то
\[
\sum_{k=1}^{\infty}c_ky_k(x)
\]
сходится равномерно к $f(x)$,
\[
y_k(x)=O(1),
\qquad
c_k=O\!\left(\frac1{k^2}\right),
\qquad k\to\infty.
\]
\end{enumerate}
\end{myremark}

\begin{mylemma}
Других собственных чисел у задачи Штурма--Лиувилля нет.
\end{mylemma}

\begin{myproof}
Пусть $y$ --- собственная функция,
\[
\mathcal Ly=\lambda y,
\qquad
\lambda\ne\lambda_k
\quad\forall k.
\]
Так как $\mathcal L$ симметричен,
\[
y\perp y_k
\qquad\forall k,
\]
следовательно,
\[
c_k=0
\qquad\forall k.
\]
По равенству Парсеваля
\[
\int_a^b y(x)^2\,\dd x
=
\sum_{k=1}^{\infty}c_k^2=0,
\]
откуда
\[
y\equiv0,
\]
что невозможно для собственной функции.
\end{myproof}

\begin{myremark}
Итоговая форма для весовой задачи.
Пусть
\[
p\in C^1[a,b],
\qquad
q,r\in C[a,b],
\qquad
p(x)>0,
\quad r(x)>0.
\]
Рассмотрим
\[
J[y]=\int_a^b(py'^2+qy^2)\,\dd x
\]
и множества
\[
M_n=
\left\{
 y\in C^1[a,b]:
 y(a)=y(b)=0,
 \int_a^b y(x)^2r(x)\,\dd x=1,
\right.
\]
\[
\left.
\hspace{35mm}
\int_a^b y(x)y_k(x)r(x)\,\dd x=0,
\quad k=1,\ldots,n-1
\right\}.
\]
Тогда
\[
\lambda_n:=\inf_{M_n}J
\]
достигается на некоторой функции $y_n$, причём
\[
\boxed{
(p y_n')'+(\lambda_nr-q)y_n=0,}
\qquad
\boxed{y_n(a)=y_n(b)=0.}
\]
Других собственных значений и собственных функций нет,
\[
\lambda_n\to+\infty.
\]
Если
\[
c_k:=\int_a^b f(x)y_k(x)r(x)\,\dd x,
\]
то
\[
\sum_{k=1}^{\infty}c_ky_k(x)=f(x)
\]
в соответствующем смысле, и
\[
\boxed{
\int_a^b f(x)^2r(x)\,\dd x
=
\sum_{k=1}^{\infty}c_k^2.}
\]
\end{myremark}

\section{Многомерные вариационные задачи}

\subsection{Уравнение Эйлера--Остроградского}

Пусть
\[
\Omega\subset\mathbb R^k
\]
--- ограниченная область и
\[
u:\overline\Omega\to\mathbb R^n.
\]
Положим
\[
X=C^1(\overline\Omega,\mathbb R^n),
\]
\[
\|u\|_X
=
\sum_{j=1}^{n}
\left(
\max_{\overline\Omega}|u_j(x)|
+
\max_{\overline\Omega}|\nabla u_j(x)|
\right).
\]
Рассмотрим функционал
\[
J[u]
=
\int_\Omega L(x,u(x),u'(x))\,\dd x,
\]
где
\[
L:\overline\Omega\times\mathbb R^n\times\mathbb R^{kn}\to\mathbb R,
\qquad
(x,u,w)\longmapsto L(x,u,w),
\]
а
\[
u'(x)
=
\left(
\frac{\partial u_j(x)}{\partial x_i}
\right)_{\substack{i=1,\ldots,k\\j=1,\ldots,n}},
\qquad
w_{ij}=\frac{\partial u_j}{\partial x_i}.
\]
Если
\[
L\in C(\overline\Omega\times\mathbb R^n\times\mathbb R^{kn}),
\]
то $J$ непрерывен на $X$.
Если
\[
L\in C^1(\overline\Omega\times\mathbb R^n\times\mathbb R^{kn}),
\]
то
\[
J\in C^1(X)
\]
и
\[
\boxed{
\delta J(u;h)
=
\int_\Omega
\left(
\sum_{j=1}^{n}
\frac{\partial L}{\partial u_j}(x,u,u')h_j(x)
+
\sum_{i=1}^{k}\sum_{j=1}^{n}
\frac{\partial L}{\partial w_{ij}}(x,u,u')
\frac{\partial h_j(x)}{\partial x_i}
\right)\dd x.}
\]

\begin{mytheorem}
Пусть
\[
L\in C^1(\overline\Omega\times\mathbb R^n\times\mathbb R^{kn}),
\]
граница $\partial\Omega$ достаточно гладкая и задана функция
\[
\varphi\in C(\partial\Omega).
\]
Пусть $u_0$ --- локальный экстремум функционала
\[
J[u]=\int_\Omega L(x,u,u')\,\dd x
\]
на множестве
\[
\left\{
 u\in C^1(\overline\Omega,\mathbb R^n):
 u\big|_{\partial\Omega}=\varphi
\right\},
\]
причём
\[
u_0\in C^2(\overline\Omega,\mathbb R^n).
\]
Тогда
\[
\boxed{
\sum_{i=1}^{k}
\frac{\partial}{\partial x_i}
\left(
\frac{\partial L(x,u_0(x),u_0'(x))}{\partial w_{ij}}
\right)
=
\frac{\partial L(x,u_0(x),u_0'(x))}{\partial u_j},
\quad j=1,\ldots,n.}
\]
\end{mytheorem}

\begin{myremark}
\begin{enumerate}
\item Производная $\dfrac{\partial}{\partial x_i}$ в уравнении берётся как полная производная, но только по переменной $x_i$.
\item Система
\[
\begin{cases}
\displaystyle
\sum_{i=1}^{k}
\frac{\partial}{\partial x_i}
\left(
\frac{\partial L}{\partial w_{ij}}
\right)
=
\dfrac{\partial L}{\partial u_j},
& j=1,\ldots,n,\\[3mm]
u_0\big|_{\partial\Omega}=\varphi
\end{cases}
\]
называется системой уравнений Эйлера--Остроградского.
\end{enumerate}
\end{myremark}

\begin{myproof}
Так как $u_0$ --- локальный экстремум, то
\[
\delta J(u_0;h)=0
\]
для всех $h\in X$ таких, что
\[
h\big|_{\partial\Omega}=0.
\]
Следовательно,
\[
\int_\Omega
\left(
\sum_{j=1}^{n}
\frac{\partial L(x,u_0,u_0')}{\partial u_j}h_j(x)
+
\sum_{i=1}^{k}\sum_{j=1}^{n}
\frac{\partial L(x,u_0,u_0')}{\partial w_{ij}}
\frac{\partial h_j(x)}{\partial x_i}
\right)\dd x
=0.
\]
Интегрируя вторую сумму по частям и используя $h|_{\partial\Omega}=0$, получаем
\[
\int_\Omega
\sum_{j=1}^{n}
\left(
\frac{\partial L(x,u_0,u_0')}{\partial u_j}
-
\sum_{i=1}^{k}
\frac{\partial}{\partial x_i}
\left(
\frac{\partial L(x,u_0,u_0')}{\partial w_{ij}}
\right)
\right)h_j(x)\,\dd x
=0.
\]
Фиксируем $j$ и берём
\[
h_1=\cdots=h_{j-1}=h_{j+1}=\cdots=h_n=0.
\]
По основной лемме вариационного исчисления
\[
\frac{\partial L(x,u_0,u_0')}{\partial u_j}
=
\sum_{i=1}^{k}
\frac{\partial}{\partial x_i}
\left(
\frac{\partial L(x,u_0,u_0')}{\partial w_{ij}}
\right),
\qquad
j=1,\ldots,n.
\]
\end{myproof}

\subsection{Естественные граничные условия}
\begin{mytheorem}
Пусть $\partial\Omega\in C^1$,
$L\in C^2(\overline\Omega\times\mathbb R^n\times\mathbb R^{kn})$,
а $u_0\in C^2(\overline\Omega,\mathbb R^n)$ --- локальный экстремум $J$ на всём пространстве
$X=C^1(\overline\Omega,\mathbb R^n)$. Тогда для $j=1,\ldots,n$
\[
\sum_{i=1}^k\frac{\partial}{\partial x_i}
 \frac{\partial L(x,u_0,u_0')}{\partial w_{ij}}
 =\frac{\partial L(x,u_0,u_0')}{\partial u_j}
 \quad\text{в }\Omega,
\]
\[
\boxed{\left.\sum_{i=1}^k
 \frac{\partial L(x,u_0,u_0')}{\partial w_{ij}}\nu_i(x)
 \right|_{\partial\Omega}=0.}
\]
Здесь $\nu$ --- внешняя единичная нормаль; последнее равенство называется естественным граничным условием.
\end{mytheorem}
\begin{myproof}
Имеем $\delta J(u_0;h)=0$ для любого $h\in X$. Сначала выбираем $h|_{\partial\Omega}=0$ и получаем уравнения Эйлера--Остроградского. Для произвольного $h$ интегрирование по частям даёт
\[
0=\int_\Omega\sum_{j=1}^n
 \left(\frac{\partial L}{\partial u_j}
 -\sum_{i=1}^k\frac{\partial}{\partial x_i}
 \frac{\partial L}{\partial w_{ij}}\right)h_j\,\dd x
 +\int_{\partial\Omega}\sum_{j=1}^n\sum_{i=1}^k
 \frac{\partial L}{\partial w_{ij}}h_j\nu_i\,\dd S.
\]
Первый интеграл равен нулю. Произвольность граничных значений $h_j$ даёт требуемые условия.
\end{myproof}
\begin{myremark}
При условном экстремуме $m$ интегральным ограничениям соответствуют постоянные множители Лагранжа $\lambda_1,\ldots,\lambda_m\in\mathbb R$.
При $m$ поточечных ограничениях множители являются функциями $\lambda_1(x),\ldots,\lambda_m(x)$.
\end{myremark}

При свободной границе естественное условие имеет вид

 \left(\frac{\partial u_j}{\partial x_i}\right)^2
 -\sum_{j=1}^3 f_j(x)u_j(x)\right)\dd x.

\section{Общая форма первой вариации}
\subsection{Интегральные функционалы на кривых}
\begin{mydef}
Параметризованная кривая класса $C^r$, $r\in\mathbb N$, --- отображение
$\gamma:[a,b]\to\mathbb R^n$ класса $C^r$ такое, что $\gamma'(x)\ne0$ для всех $x\in[a,b]$.
\end{mydef}
\begin{mydef}
Кривые $\gamma_1:[a_1,b_1]\to\mathbb R^n$ и $\gamma_2:[a_2,b_2]\to\mathbb R^n$ эквивалентны, если существует биекция
$\chi:[a_1,b_1]\to[a_2,b_2]$, $\chi,\chi^{-1}\in C^r$, $\chi'>0$, такая, что $\gamma_1=\gamma_2\circ\chi$.
Ориентированная кривая --- класс эквивалентности таких параметризованных кривых.
\end{mydef}
\begin{myexe}
Проверить, что $\sim$ является отношением эквивалентности.
\end{myexe}
\begin{myremark}
У эквивалентных кривых совпадают образы, начальные и конечные точки, а также длины:
\[
\operatorname{Im}\gamma_1=\operatorname{Im}\gamma_2,\quad
\gamma_1(a_1)=\gamma_2(a_2),\quad\gamma_1(b_1)=\gamma_2(b_2),\quad
\ell(\gamma)=\int_a^b|\gamma'(x)|\,\dd x.
\]
\end{myremark}
Пусть $F\in C(\mathbb R^n\times\mathbb R^n)$ положительно однородна степени один по второму аргументу:
\[
F(z,\lambda w)=\lambda F(z,w),\qquad\lambda>0.
\]
Рассмотрим $J[\gamma]=\int_a^b F(\gamma(x),\gamma'(x))\,\dd x$.
% Рукопись, с. 24.
\begin{mylemma}
Функционал $J$ не зависит от выбора параметризации ориентированной кривой.
\end{mylemma}
\begin{myproof}
Если $\gamma_1(x)=\gamma_2(\chi(x))$, то
\[\begin{array}{@{}>{\displaystyle}r@{}>{\displaystyle}l@{}}
J[\gamma_1]
&=\int_{a_1}^{b_1}F(\gamma_2(\chi(x)),\chi'(x)\gamma_2'(\chi(x)))\,\dd x\\
&=\int_{a_1}^{b_1}F(\gamma_2(\chi(x)),\gamma_2'(\chi(x)))\chi'(x)\,\dd x\\
&=\int_{a_2}^{b_2}F(\gamma_2(s),\gamma_2'(s))\,\dd s=J[\gamma_2].
\end{array}\]
\end{myproof}
\begin{myexample}
Для $F(z,w)=f(z)|w|$, $f\in C(\mathbb R^n)$,
\[
J[\gamma]=\int_a^b f(\gamma(x))|\gamma'(x)|\,\dd x=\int_\gamma f\,\dd s.
\]
При $f\equiv1$ это длина кривой.
\end{myexample}
\begin{myexample}
Пусть $\gamma(x)=(x,y(x))$, $y:[a,b]\to\mathbb R^m$, $n=m+1$.
Для $z=(z_1,\widehat z)$, $w=(w_1,\widehat w)$, $w_1>0$, положим
\[
F(z,w)=L\left(z_1,\widehat z,\frac{\widehat w}{w_1}\right)w_1.
\]
Тогда $F(z,\lambda w)=\lambda F(z,w)$ и
\[
J[\gamma]=\int_a^b F(\gamma,\gamma')\,\dd x
 =\int_a^b L(x,y(x),y'(x))\,\dd x.
\]
\end{myexample}
\begin{myexe}
Для $F\in C^1(\mathbb R^n\setminus\{0\})$ доказать эквивалентность
\[
F(\lambda w)=\lambda F(w)\quad(\lambda>0)
\quad\Longleftrightarrow\quad
F(w)=\langle\nabla F(w),w\rangle.
\]
\end{myexe}
\subsection{Общая формула первой вариации}
\begin{mytheorem}
Пусть $F\in C^2(\mathbb R^n\times(\mathbb R^n\setminus\{0\}))$,
$F(z,\lambda w)=\lambda F(z,w)$, а
$\Gamma:[a,b]\times[\alpha_1,\alpha_2]\to\mathbb R^n$ --- гладкое семейство регулярных кривых.
Обозначим
\[
\Phi(\alpha)=\int_a^b F(\Gamma(x,\alpha),\Gamma_x'(x,\alpha))\,\dd x,
\]
\[
\mathcal E_F\{\gamma\}(x)=\nabla_zF(\gamma(x),\gamma'(x))
 -\frac{\dd}{\dd x}\nabla_wF(\gamma(x),\gamma'(x)).
\]
Тогда
\[
\boxed{\Phi'(\alpha)=\int_a^b
 \langle\mathcal E_F\{\Gamma(\cdot,\alpha)\},\Gamma_\alpha'\rangle\,\dd x
 +\left.\langle\nabla_wF(\Gamma,\Gamma_x'),\Gamma_\alpha'\rangle\right|_a^b.}
\]
\end{mytheorem}
\begin{myproof}
Дифференцируем под знаком интеграла и интегрируем по частям:
\[\begin{array}{@{}>{\displaystyle}r@{}>{\displaystyle}l@{}}
\Phi'(\alpha)&=\int_a^b\left(
 \langle\nabla_zF,\Gamma_\alpha'\rangle+
 \langle\nabla_wF,\Gamma_{x\alpha}''\rangle\right)\dd x\\
&=\int_a^b\left\langle\nabla_zF-\frac{\dd}{\dd x}\nabla_wF,
 \Gamma_\alpha'\right\rangle\dd x
 +\left.\langle\nabla_wF,\Gamma_\alpha'\rangle\right|_a^b.
\end{array}\]
\end{myproof}
\begin{mylemma}
Из однородности $F$ следуют равенства
\[
\nabla_zF(z,\lambda w)=\lambda\nabla_zF(z,w),\qquad
\nabla_wF(z,\lambda w)=\nabla_wF(z,w),\quad\lambda>0.
\]
\end{mylemma}
\begin{myproof}
Первое равенство получается дифференцированием по $z$. Дифференцирование по $w_j$ даёт
$\lambda F_{w_j}(z,\lambda w)=\lambda F_{w_j}(z,w)$.
\end{myproof}

\mathcal E_F\{\gamma_1\}(x)=\chi'(x)\mathcal E_F\{\gamma_2\}(\chi(x)).

\subsection{Уравнение Эйлера}
\begin{mytheorem}
Пусть $F$ удовлетворяет указанным выше условиям. Если $\gamma_0$ --- локальный экстремум $J$ на множестве регулярных $C^1$-кривых с
$\gamma(a)=y_1$, $\gamma(b)=y_2$, и $\gamma_0\in C^2$, то
\[
\boxed{\mathcal E_F\{\gamma_0\}=0,\qquad
\nabla_zF(\gamma_0,\gamma_0')-\frac{\dd}{\dd x}\nabla_wF(\gamma_0,\gamma_0')=0.}
\]
\end{mytheorem}
\begin{myproof}
Берём $\Gamma(x,\alpha)=\gamma_0(x)+\alpha h(x)$, где $h\in C^1$, $h(a)=h(b)=0$.
Так как $\gamma_0'\ne0$ на компакте, существует $\delta_0>0$:
$|\gamma_0'|\ge\delta_0$. Поэтому при достаточно малых $|\alpha|$ все кривые $\Gamma(\cdot,\alpha)$ регулярны.
Точка $\alpha=0$ --- локальный экстремум $\Phi$, следовательно,
\[
0=\Phi'(0)=\int_a^b\langle\mathcal E_F\{\gamma_0\}(x),h(x)\rangle\,\dd x.
\]
Основная лемма вариационного исчисления завершает доказательство.
\end{myproof}
\begin{myremark}
Условие $\mathcal E_F\{\gamma\}=0$ не зависит от параметризации.
Кривая, удовлетворяющая уравнению Эйлера, называется экстремалью.
\end{myremark}

\subsection{Условия трансверсальности}
\begin{myexe}
Построить $\theta\in C^\infty(\mathbb R)$, равную единице в окрестности $a$ и нулю в окрестности $b$.
\end{myexe}
\begin{mytheorem}
Пусть $M_1,M_2\subset\mathbb R^n$ --- подмногообразия класса $C^1$,
а $\gamma_0\in C^2$ --- локальный экстремум $J$ на множестве регулярных кривых с
$\gamma(a)\in M_1$, $\gamma(b)\in M_2$. Тогда
\[
\mathcal E_F\{\gamma_0\}=0,
\]
\[
\boxed{\nabla_wF(\gamma_0(a),\gamma_0'(a))\perp T_{y_1}M_1,\qquad
\nabla_wF(\gamma_0(b),\gamma_0'(b))\perp T_{y_2}M_2,}
\qquad y_1=\gamma_0(a),\quad y_2=\gamma_0(b).
\]
Это условия трансверсальности.
\end{mytheorem}
\begin{myproof}
Вариации с закреплёнными концами дают уравнение Эйлера.
Пусть $u:(-\varepsilon,\varepsilon)\to M_1$, $u\in C^1$, $u(0)=y_1$.
Положим
\[
\Gamma(x,\alpha)=\gamma_0(x)+\theta(x)(u(\alpha)-y_1).
\]
При малых $|\alpha|$ имеем $\Gamma_x'\ne0$, $\Gamma(a,\alpha)=u(\alpha)$,
$\Gamma(b,\alpha)=y_2$ и $\Gamma_\alpha'=\theta(x)u'(\alpha)$.
Общая формула первой вариации даёт
\[
0=\Phi'(0)=-\langle\nabla_wF(\gamma_0(a),\gamma_0'(a)),u'(0)\rangle.
\]
Векторы $u'(0)$ заполняют $T_{y_1}M_1$. Для второго конца рассуждение аналогично.
\end{myproof}
\begin{myremark}
Вектор $\nabla_wF(\gamma,\gamma')$ не зависит от выбора параметризации.
\end{myremark}
\begin{myexe}

\subsection{Замена переменных. Теорема Нётер}
Пусть $T:\mathbb R^n\to\mathbb R^n$, $T\in C^2$, $\det T'(z)\ne0$.
Обозначим
\[
\gamma_T=T\circ\gamma,\qquad F_T(z,w)=F(T(z),T'(z)w),\qquad
J_T[\gamma]=\int_a^b F_T(\gamma,\gamma')\,\dd x.
\]
По правилу дифференцирования сложной функции
\[
\frac{\dd}{\dd x}T(\gamma(x))=T'(\gamma(x))\gamma'(x),\qquad
J[\gamma_T]=J_T[\gamma].
\]
\begin{myexe}
Доказать, что
$\mathcal E_F\{\gamma_T\}=0\Longleftrightarrow\mathcal E_{F_T}\{\gamma\}=0$.
\end{myexe}
\begin{mytheorem}
\textbf{(Нётер.)} Пусть $T_\alpha:\mathbb R^n\to\mathbb R^n$ --- семейство замен переменных,
$\det T_\alpha'(z)\ne0$, отображение $(\alpha,z)\mapsto T_\alpha(z)$ класса $C^2$ и $T_0(z)=z$.
Пусть $F\in C^2(\mathbb R^n\times(\mathbb R^n\setminus\{0\}))$ однородна степени один по $w$,
а $J_{T_\alpha}=J$ для всех $\alpha$ (в частности, достаточно $F_{T_\alpha}=F$).
Тогда вдоль любой экстремали $\gamma$
\[
\boxed{\left\langle\nabla_wF(\gamma(x),\gamma'(x)),
 \left.\frac{\partial T_\alpha(\gamma(x))}{\partial\alpha}\right|_{\alpha=0}
 \right\rangle=\mathrm{const}.}
\]
\end{mytheorem}
\begin{myproof}
Фиксируем $a<c\le b$ и ограничение $\gamma_c$ на $[a,c]$.
Положим $\Gamma(x,\alpha)=T_\alpha(\gamma_c(x))$.
Тогда $\Gamma_x'=T_\alpha'(\gamma_c)\gamma_c'\ne0$, а $J[\Gamma(\cdot,\alpha)]$ не зависит от $\alpha$.
Следовательно,
\[
0=\int_a^c\langle\mathcal E_F\{\Gamma\},\Gamma_\alpha'\rangle\,\dd x
 +\left.\langle\nabla_wF(\Gamma,\Gamma_x'),\Gamma_\alpha'\rangle\right|_a^c.
\]
При $\alpha=0$ имеем $\Gamma=\gamma_c$ и $\mathcal E_F\{\gamma\}=0$.
Значит, выражения в рамке в точках $a$ и $c$ совпадают. Точка $c$ произвольна.
\end{myproof}
\begin{myexample}
Пусть $\gamma(t)=(t,r(t))$, $r(t)\in\mathbb R^N$, $n=N+1$,
\[
J[\gamma]=\int_a^b L(t,r(t),\dot r(t))\,\dd t,
\qquad F(z,w)=L\left(z_1,\widehat z,\frac{\widehat w}{w_1}\right)w_1.
\]
Если $L(t,r,v)=L(r,v)$, функционал инвариантен относительно сдвигов времени
$T_\alpha(z)=(z_1-\alpha,\widehat z)$, $T_\alpha(t,r)=(t-\alpha,r)$.
При этом
\[
F_{w_1}=L-\left\langle\frac{\widehat w}{w_1},\nabla_vL\right\rangle,
\quad\nabla_{\widehat w}F=\nabla_vL,
\quad\frac{\partial T_\alpha}{\partial\alpha}=(-1,0).
\]
По теореме Нётер
\[
\boxed{\langle\dot r(t),\nabla_vL(r(t),\dot r(t))\rangle-L(r(t),\dot r(t))
 =\mathrm{const},}
\]
что является законом сохранения энергии.
\end{myexample}
\begin{myexe}
Получить из теоремы Нётер: 1) закон сохранения импульса из однородности пространства;
2) закон сохранения момента импульса из изотропности пространства.
\end{myexe}

\section{Уравнения Гамильтона}
\subsection{Преобразование Лежандра}
Пусть $F:\mathbb R^n\to\mathbb R$, $F\in C^2$, и
$\nabla F:\mathbb R^n\to\mathbb R^n$ --- биекция.
\begin{mydef}
Преобразованием Лежандра функции $F$ называется функция
\[
G(p)=\langle v,p\rangle-F(v),\qquad \nabla F(v)=p.
\]
Если $\Phi=(\nabla F)^{-1}$, то $G(p)=\langle\Phi(p),p\rangle-F(\Phi(p))$.
При условии $\det(\nabla F)'(v)\ne0$ для всех $v$ имеем $\Phi\in C^1$.
\end{mydef}
\begin{myprop}
\begin{enumerate}
\item $\nabla G(p)=v=\Phi(p)$.
\item Преобразование Лежандра $G$ равно $F$.
\item При преобразовании по части переменных, если
$G(y,p)=\langle\Phi(y,p),p\rangle-F(y,\Phi(y,p))$, то
\[
\nabla_yG(y,p)=-\nabla_yF(y,v),\qquad v=\Phi(y,p).
\]
\end{enumerate}
\end{myprop}
\begin{myproof}
\[
\frac{\partial G}{\partial p_i}=
 \sum_{j=1}^n\frac{\partial\Phi_j}{\partial p_i}p_j+\Phi_i
 -\sum_{j=1}^n\frac{\partial F}{\partial v_j}(\Phi(p))
 \frac{\partial\Phi_j}{\partial p_i}=\Phi_i(p).
\]
Поэтому обратное преобразование даёт $\langle p,v\rangle-G(p)=F(v)$.
Аналогично,
\[
\frac{\partial G}{\partial y_i}=
 \sum_j\frac{\partial\Phi_j}{\partial y_i}p_j
 -\frac{\partial F}{\partial y_i}(y,\Phi)
 -\sum_j\frac{\partial F}{\partial v_j}(y,\Phi)
 \frac{\partial\Phi_j}{\partial y_i}
 =-\frac{\partial F}{\partial y_i}(y,v).
\]
\end{myproof}
\subsection{Выпуклые функции}
\begin{mydef}
Функция $F:\mathbb R^n\to\mathbb R$ называется выпуклой, если в каждой точке $v_0$ существует опорная аффинная функция $\ell$:
\[
\ell(v_0)=F(v_0),\qquad F(v)\ge\ell(v)\quad\forall v.
\]
То есть существует $k\in\mathbb R^n$ такое, что
\[
F(v)\ge F(v_0)+\langle k,v-v_0\rangle\quad\forall v\in\mathbb R^n.
\]
\end{mydef}
\begin{mylemma}
Пусть $\{\ell_\beta\}$ --- семейство аффинных функций и
$F(v)=\sup_\beta\ell_\beta(v)<+\infty$ для всех $v$. Тогда $F$ выпукла.
\end{mylemma}
\begin{myproof}
Фиксируем $v_0$. Если супремум достигается на $\ell_\beta$, эта функция опорная.
Иначе выбираем $\beta_m$ так, что $\ell_{\beta_m}(v_0)\to F(v_0)$, и пишем
\[
\ell_{\beta_m}(v)=\ell_{\beta_m}(v_0)+\langle k_m,v-v_0\rangle.
\]
Последовательность $k_m$ ограничена: подстановка $v=v_0\pm e_j$ ограничивает каждую её координату.
Извлекаем подпоследовательность $k_{m_j}\to k_*$.
Переходя к пределу в неравенстве $F(v)\ge\ell_{\beta_{m_j}}(v)$, получаем
$F(v)\ge F(v_0)+\langle k_*,v-v_0\rangle$.
\end{myproof}
\begin{myexe}
Пусть $F$ выпукла, $\alpha_j\ge0$, $\sum_{j=1}^m\alpha_j=1$. Доказать
\[
F\left(\sum_{j=1}^m\alpha_jv_j\right)\le\sum_{j=1}^m\alpha_jF(v_j).
\]
\end{myexe}
\begin{mydef}
Преобразование Лежандра выпуклой функции $F$ определяется формулой
\[
\boxed{G(p)=\sup_{v\in\mathbb R^n}(\langle v,p\rangle-F(v)).}
\]
Возможно значение $G(p)=+\infty$.
\end{mydef}
\begin{myprop}
\begin{enumerate}
\item $G(p)\ge-F(0)$.
\item Если $G$ всюду конечна, она выпукла.
\item $\langle v,p\rangle\le F(v)+G(p)$ для всех $v,p$.
\item Если $G$ всюду конечна, её преобразование Лежандра равно $F$.
\end{enumerate}
\end{myprop}
\begin{myproof}
Первые три свойства непосредственно следуют из определения и леммы.
Пусть $H(v_0)=\sup_p(\langle v_0,p\rangle-G(p))$. Из третьего свойства $H(v_0)\le F(v_0)$.
По выпуклости найдётся $p_0$ такое, что
\[
F(v)\ge F(v_0)+\langle p_0,v-v_0\rangle.
\]
Отсюда $G(p_0)\le\langle v_0,p_0\rangle-F(v_0)$ и
\[
H(v_0)\ge\langle v_0,p_0\rangle-G(p_0)\ge F(v_0).
\]
Таким образом, $H=F$; в частности, $G(p_0)=\langle v_0,p_0\rangle-F(v_0)$.
\end{myproof}
\begin{myremark}
Для гладкой выпуклой $F$ оба определения согласованы: если
$G(p_0)=\langle v_0,p_0\rangle-F(v_0)$, то условие стационарности по $v$ даёт
$p_0=\nabla F(v_0)$.
Преобразование можно определять также на областях $\Omega_1,\Omega_2$, если
$\nabla F:\Omega_1\to\Omega_2$ --- биекция.
\end{myremark}
\begin{myexample}
Пусть $n=1$, $\Omega=(0,+\infty)$, $1<q,q'<\infty$, $1/q+1/q'=1$.
Для $F(v)=v^q/q$ имеем $F'(v)=v^{q-1}$, $F''(v)=(q-1)v^{q-2}>0$.
Из $v^{q-1}=p$ получаем $v=p^{1/(q-1)}$ и
\[
G(p)=vp-F(v)=\frac{p^{q'}}{q'}.
\]
Неравенство $vp\le F(v)+G(p)$ даёт неравенство Юнга
\[
xy\le\frac{x^q}{q}+\frac{y^{q'}}{q'},\qquad x,y>0.
\]
\end{myexample}
\begin{myexample}
Пусть $A=A^{\mathsf T}$ --- невырожденная вещественная матрица,
$F(v)=\tfrac12\langle Av,v\rangle$. Тогда $\nabla F(v)=Av$, $v=A^{-1}p$,
\[
G(p)=\langle A^{-1}p,p\rangle
 -\frac12\langle AA^{-1}p,A^{-1}p\rangle
 =\frac12\langle A^{-1}p,p\rangle.
\]
Функция $F$ выпукла тогда и только тогда, когда все собственные значения $A$ неотрицательны.
\end{myexample}
\begin{myexe}
Вычислить преобразования Лежандра: 1) $e^x$ на $\mathbb R$;
2) $x^4+y^4$ и $(x^2+y^2)^2$ на $\mathbb R^2$.
\end{myexe}
\subsection{Канонические уравнения Гамильтона}
\begin{mydef}
Пусть $H\in C^1([a,b]\times\mathbb R^n\times\mathbb R^n)$.
Системой уравнений Гамильтона для функций $f,g:[a,b]\to\mathbb R^n$ называется
\[
\boxed{\begin{cases}
\dot f(t)=\nabla_pH(t,f(t),g(t)),\\
\dot g(t)=-\nabla_qH(t,f(t),g(t)).
\end{cases}}
\]
\end{mydef}
\begin{mytheorem}
Пусть $L,H\in C^2([a,b]\times\mathbb R^n\times\mathbb R^n)$ и
для каждого $(t,q)$ функции $L(t,q,v)$ и $H(t,q,p)$ являются преобразованиями Лежандра друг друга.
Тогда для $f\in C^1([a,b],\mathbb R^n)$ эквивалентны:
\begin{enumerate}
\item $\displaystyle\frac{\dd}{\dd t}\nabla_vL(t,f(t),\dot f(t))=\nabla_qL(t,f(t),\dot f(t))$;
\item существует $g\in C^1([a,b],\mathbb R^n)$, для которого $(f,g)$ удовлетворяет уравнениям Гамильтона.
\end{enumerate}
\end{mytheorem}
\begin{myproof}
Из первого утверждения положим $g(t)=\nabla_vL(t,f(t),\dot f(t))$.
Соотношения преобразования Лежандра дают
\[
\nabla_vL(t,q,v)=p\ \Longleftrightarrow\ \nabla_pH(t,q,p)=v,
\qquad\nabla_qL(t,q,v)=-\nabla_qH(t,q,p).
\]
Следовательно, $\dot f=\nabla_pH(t,f,g)$ и
$\dot g=\frac{\dd}{\dd t}\nabla_vL=\nabla_qL=-\nabla_qH(t,f,g)$.
Обратно, из уравнений Гамильтона получаем
$\nabla_vL(t,f,\dot f)=g$ и
\[
\frac{\dd}{\dd t}\nabla_vL(t,f,\dot f)=\dot g
 =-\nabla_qH(t,f,g)=\nabla_qL(t,f,\dot f).
\]
\end{myproof}
\begin{myremark}
Вдоль решения
$H(t,f,g)=\langle g(t),\dot f(t)\rangle-L(t,f(t),\dot f(t))$.
\end{myremark}

\section{Уравнения в частных производных первого порядка}
\subsection{Характеристики}
Рассмотрим уравнение
\[
\mathcal F(x,u(x),\nabla u(x))=0,\qquad
\mathcal F:\mathbb R^n\times\mathbb R\times\mathbb R^n\to\mathbb R.
\]
Пусть $\Gamma$ --- гиперповерхность в $\mathbb R^n$,
$\dim\Gamma=n-1$, и $u|_\Gamma=\varphi$.
Для кривой $x(t)$, $x(0)=x_0\in\Gamma$, положим $z(t)=u(x(t))$, $p(t)=\nabla u(x(t))$.
Тогда
\[
\dot p_j=\sum_k u_{x_jx_k}\dot x_k,\qquad
\mathcal F_{x_j}+\mathcal F_z u_{x_j}+\sum_k\mathcal F_{p_k}u_{x_kx_j}=0.
\]
Выбирая $\dot x_k=\mathcal F_{p_k}$, получаем
$\dot p_j=-\mathcal F_{x_j}-\mathcal F_zp_j$, а
$\dot z=\sum_kp_k\dot x_k$.
\begin{mydef}
Характеристической системой уравнения $\mathcal F(x,u,\nabla u)=0$ называется система
\[
\boxed{\begin{cases}
\dot x=\nabla_p\mathcal F(x,z,p),\\
\dot z=\langle p,\nabla_p\mathcal F(x,z,p)\rangle,\\
\dot p=-\nabla_x\mathcal F(x,z,p)-\mathcal F_z(x,z,p)p.
\end{cases}}\tag{Х}
\]
Её интегральная кривая $(x(t),z(t),p(t))$ в $\mathbb R^{2n+1}$ называется характеристикой,
а $x(t)$ --- проекцией характеристики в $\mathbb R^n$.
\end{mydef}
\begin{myremark}
Функция $\mathcal F$ постоянна на характеристиках:
\[
\frac{\dd}{\dd t}\mathcal F(x,z,p)
=\langle\nabla_x\mathcal F,\nabla_p\mathcal F\rangle
+\mathcal F_z\langle p,\nabla_p\mathcal F\rangle
-\langle\nabla_p\mathcal F,\nabla_x\mathcal F+\mathcal F_zp\rangle=0.
\]
\end{myremark}
\subsection{Начальные условия. Допустимые тройки}
Пусть $\Gamma=\{x=(x_1,\ldots,x_{n-1},0)\}$. На ней задаём
\[
x(0)=x^0\in\Gamma,\quad z(0)=\varphi(x^0),\quad
p_i(0)=\varphi_{x_i}(x^0),\ i=1,\ldots,n-1,
\]
\[
\mathcal F(x^0,\varphi(x^0),p(0))=0.
\]
\begin{mydef}
Тройка $(x^0,z^0,p^0)$ называется допустимой, если
\[
z^0=\varphi(x^0),\quad p_i^0=\varphi_{x_i}(x^0)\ (i<n),\quad
\mathcal F(x^0,z^0,p^0)=0.
\]
\end{mydef}
\begin{mylemma}
Пусть $(x^0,z^0,p^0)$ --- допустимая тройка, $\mathcal F\in C^1$, $\varphi\in C^2$ и
$\mathcal F_{p_n}(x^0,z^0,p^0)\ne0$.
Тогда существует окрестность $W$ точки $x^0$ на $\Gamma$ и $q\in C^1(W,\mathbb R^n)$ такие, что
\[
q_i(y)=\varphi_{y_i}(y)\ (i<n),\qquad
\mathcal F(y,\varphi(y),q(y))=0,\qquad q(x^0)=p^0.
\]
\end{mylemma}
\begin{myproof}
Положим
\[
G(y,w)=\mathcal F\bigl(y,\varphi(y),
 (\varphi_{y_1}(y),\ldots,\varphi_{y_{n-1}}(y),w)\bigr).
\]
Имеем $G_w(x^0,p_n^0)=\mathcal F_{p_n}(x^0,z^0,p^0)\ne0$.
По теореме о неявной функции существует $g\in C^1(W)$:
$G(y,w)=0\Longleftrightarrow w=g(y)$ в соответствующей окрестности.
Берём $q_n=g$ и $q_i=\varphi_{y_i}$ при $i<n$.
\end{myproof}
\begin{myremark}
Условие $\mathcal F_{p_n}\ne0$ означает нехарактеристичность начальной гиперповерхности:
$\langle\nabla_p\mathcal F,\nu\rangle\ne0$.
\end{myremark}
\subsection{Решение характеристической системы}
Для допустимых троек $(y,\varphi(y),q(y))$ решим (Х) с начальными условиями
\[
x(y,0)=y,\qquad z(y,0)=\varphi(y),\qquad p(y,0)=q(y).
\]
\begin{mylemma}
На области определения решения $\mathcal F(x(y,t),z(y,t),p(y,t))=0$.
\end{mylemma}
\begin{myproof}
Производная по $t$ равна нулю, а при $t=0$ равенство следует из допустимости тройки.
\end{myproof}
Пусть теперь $\mathcal F\in C^2$, $\varphi\in C^2$ и $q\in C^1$.
Решение $(x,z,p)$ класса $C^1$ по $(y,t)$; необходимые смешанные производные по $y,t$ существуют.
\begin{mylemma}
Имеем
\[
\frac{\partial z}{\partial t}=\sum_{k=1}^n p_k\frac{\partial x_k}{\partial t},\qquad
\frac{\partial z}{\partial y_i}=\sum_{k=1}^n p_k\frac{\partial x_k}{\partial y_i},
\quad i=1,\ldots,n-1.
\]
\end{mylemma}
\begin{myproof}
Первое равенство --- второе уравнение (Х). Для фиксированных $y,i$ положим
\[
r(t)=z_{y_i}(y,t)-\sum_kp_k(y,t)x_{k,y_i}(y,t).
\]
При $t=0$ получаем
$r(0)=\varphi_{y_i}(y)-\sum_kq_k(y)\delta_{ki}=\varphi_{y_i}-q_i=0$.
Дифференцируя первое равенство по $y_i$ и выражение для $r$ по $t$, находим
\[
\dot r=\sum_k(p_{k,y_i}x_{k,t}-p_{k,t}x_{k,y_i})
=\sum_k\bigl(\mathcal F_{p_k}p_{k,y_i}
 +(\mathcal F_{x_k}+\mathcal F_zp_k)x_{k,y_i}\bigr).
\]
Но дифференцирование $\mathcal F(x,z,p)=0$ по $y_i$ даёт
\[
\sum_k\mathcal F_{x_k}x_{k,y_i}+\mathcal F_z z_{y_i}
 +\sum_k\mathcal F_{p_k}p_{k,y_i}=0.
\]
Следовательно, $\dot r=-\mathcal F_zr$, $r(0)=0$.
По теореме единственности $r\equiv0$.
\end{myproof}
\subsection{Локальная обратимость}
\begin{mylemma}
При $\mathcal F_{p_n}(x^0,z^0,p^0)\ne0$ отображение $(y,t)\mapsto x(y,t)$ локально обратимо в точке $(x^0,0)$.
Существуют окрестность $V$ точки $x^0$ в $\mathbb R^n$ и $h>0$ такие, что
каждый $x\in V$ единственным образом представляется в виде $x=x(y,t)$ с $(y,t)$ в соответствующей окрестности $W\times(-h,h)$.
\end{mylemma}
\begin{myproof}
При $t=0$ имеем $x_{j,y_i}=\delta_{ji}$, $i<n$, $j\le n$,
а $x_{j,t}(x^0,0)=\mathcal F_{p_j}(x^0,z^0,p^0)$.
Поэтому
\[
\det\frac{\partial x}{\partial(y,t)}(x^0,0)
=\det\begin{pmatrix}
1&\cdots&0&\mathcal F_{p_1}\\
\vdots&\ddots&\vdots&\vdots\\
0&\cdots&1&\mathcal F_{p_{n-1}}\\
0&\cdots&0&\mathcal F_{p_n}
\end{pmatrix}=\mathcal F_{p_n}\ne0.
\]
Применяем теорему об обратном отображении.
\end{myproof}

\subsection{Локальная теорема существования}
\begin{mytheorem}
Пусть $\mathcal F\in C^2(\mathbb R^{2n+1})$, $\varphi\in C^2(\Gamma)$,
$(x^0,z^0,p^0)$ --- допустимая тройка и $\mathcal F_{p_n}(x^0,z^0,p^0)\ne0$.
Построим $q$ и семейство характеристик, как выше.
Пусть $(y(x),t(x))$ --- локальное обратное отображение к $(y,t)\mapsto x(y,t)$.
Тогда в некоторой окрестности $V$ точки $x^0$ функция
\[
\boxed{u(x)=z(y(x),t(x))}
\]
решает задачу
\[
\mathcal F(x,u(x),\nabla u(x))=0,\qquad u|_{\Gamma\cap V}=\varphi.
\]
\end{mytheorem}
\begin{myproof}
Для $y_0\in\Gamma\cap V$ имеем $x(y_0,0)=y_0$, поэтому $u(y_0)=z(y_0,0)=\varphi(y_0)$.
Уже известно, что $\mathcal F(x,u(x),p(y(x),t(x)))=0$.
Остаётся проверить $\nabla u=p$. По предыдущей лемме
\[\begin{array}{@{}>{\displaystyle}r@{}>{\displaystyle}l@{}}
\frac{\partial u}{\partial x_j}
&=\sum_{i=1}^{n-1}z_{y_i}y_{i,x_j}+z_t t_{x_j}\\
&=\sum_kp_k\left(\sum_{i=1}^{n-1}x_{k,y_i}y_{i,x_j}+x_{k,t}t_{x_j}\right)
 =\sum_kp_k\delta_{kj}=p_j.
\end{array}\]
Здесь выражение в скобках --- производная тождественного отображения $x\mapsto x(y(x),t(x))$.
\end{myproof}
\subsection{Квазилинейные уравнения первого порядка}
Для однородного уравнения
\[
\langle a(x,u(x)),\nabla u(x)\rangle=0
\]
берём $\mathcal F(x,z,p)=\langle a(x,z),p\rangle$.
Тогда $\dot x=a(x,z)$, $\dot z=\langle a,\nabla u\rangle=0$,
то есть $z=\varphi(y)$ постоянно и достаточно решить $n$ уравнений
\[
\dot x(t)=a(x(t),\varphi(y)),\qquad x(0)=y.
\]
Иногда характеристики записывают в симметрической форме
$\dd x_1/a_1=\cdots=\dd x_n/a_n$.
\begin{myexample}
При $n=2$ уравнение
\[
x_1u_{x_2}-x_2u_{x_1}=0
\]

Отсюда $\ddot x_1=-x_1$,
\[
x_1=A\cos t+B\sin t,\qquad x_2=A\sin t-B\cos t,\qquad
x_1^2+x_2^2=A^2+B^2.
\]
Решения имеют вид $u(x)=f(x_1^2+x_2^2)$.
Условие нехарактеристичности начальной кривой означает, что её нормаль не перпендикулярна вектору $(-x_2,x_1)$;
эквивалентно, её касательная не параллельна этому вектору.
\end{myexample}

$N^{-3/2}\int_{\mathbb R}e^{-s^2}s^2\,\dd s$.
\item Если максимум достигается в $c=b$, $\varphi'(b)=0$, $\varphi''(b)<0$, то
\[
I(N)=\frac12\sqrt{\frac{2\pi}{-N\varphi''(b)}}e^{N\varphi(b)}F(b)
 +O\left(\frac{e^{N\varphi(b)}}N\right).
\]
\item Если имеется несколько равных абсолютных максимумов $c_1,\ldots,c_\ell$,
главные вклады складываются: $I\sim I^{(c_1)}+\cdots+I^{(c_\ell)}$.
Максимумы меньшей высоты дают экспоненциально меньшие вклады.
\end{enumerate}
\end{myremark}
\begin{myexe}
Пусть $c$ --- единственный максимум, $\varphi'(c)=\varphi''(c)=\varphi'''(c)=0$,
$\varphi^{(4)}(c)<0$. Найти главную асимптотику $I(N)$.
\end{myexe}

\section{Метод стационарной фазы}
Пусть
\[
I(N)=\int_a^b e^{iN\varphi(x)}F(x)\,\dd x,
\qquad\varphi,F\in C^\infty[a,b],\quad\varphi(x)\in\mathbb R.
\]
\subsection{Нестационарная фаза}
Если $\varphi'\ne0$ на $[a,b]$, делаем замену $t=\varphi(x)$,
$\widetilde F(t)=F(x)/\varphi'(x)$, где $x=\varphi^{-1}(t)$,
$\widetilde a=\varphi(a)$, $\widetilde b=\varphi(b)$.
Тогда $I(N)=\int_{\widetilde a}^{\widetilde b}e^{iNt}\widetilde F(t)\,\dd t$ и
\[\begin{array}{@{}>{\displaystyle}r@{}>{\displaystyle}l@{}}
I(N)&=\left.\frac{e^{iNt}\widetilde F(t)}{iN}\right|_{\widetilde a}^{\widetilde b}
 -\frac1{iN}\int_{\widetilde a}^{\widetilde b}e^{iNt}\widetilde F'(t)\,\dd t\\
&\sim\sum_{k=0}^\infty\frac{(-1)^k}{(iN)^{k+1}}
 \bigl(e^{iN\widetilde b}\widetilde F^{(k)}(\widetilde b)
       -e^{iN\widetilde a}\widetilde F^{(k)}(\widetilde a)\bigr).
\end{array}\]
При возвращении к $x$ используется $\dd/\dd t=(\varphi')^{-1}\dd/\dd x$.
\begin{mylemma}
Если $\varphi'\ne0$ на $[a,b]$, то
\[
I(N)=\frac{e^{iN\varphi(b)}F(b)}{iN\varphi'(b)}
 -\frac{e^{iN\varphi(a)}F(a)}{iN\varphi'(a)}+O(N^{-2}).
\]
\end{mylemma}
\begin{mylemma}
Если $\varphi'\ne0$ на $[a,b]$, $F\in C_0^\infty(a,b)$, то
$I(N)=O(N^{-m})$ для любого $m$.
\end{mylemma}
\begin{myproof}
Последовательно интегрируем по частям. В первом случае два интегрирования дают заявленный остаток.
Во втором все граничные члены исчезают, и число интегрирований можно выбрать произвольно большим.
\end{myproof}
\subsection{Леммы о квадратичной фазе}
\begin{mylemma}
Пусть $a>\delta>0$, $f\in C_0^\infty(-a,a)$ и $f\equiv1$ на $[-\delta,\delta]$.
Тогда для любого $m$
\[
\int_{-a}^a e^{iNt^2}f(t)\,\dd t
 =\sqrt{\frac\pi N}e^{i\pi/4}+O(N^{-m}).
\]
\end{mylemma}
\begin{myproof}
Деформируем центральный отрезок контура на отрезок прямой $t=e^{i\pi/4}x$,
$-\delta/\sqrt2\le x\le\delta/\sqrt2$, соединив его концы вертикальными отрезками с точками $\pm\delta$.
В центральной части $f=1$, поэтому деформация разрешена аналитичностью экспоненты.
На наклонном отрезке
\[
\int e^{iNt^2}\,\dd t
 =e^{i\pi/4}\int_{-\delta/\sqrt2}^{\delta/\sqrt2}e^{-Nx^2}\,\dd x
 =e^{i\pi/4}\sqrt{\frac\pi N}+o(e^{-N\delta^2/2}).
\]
Оставшиеся части контура состоят из внешних вещественных отрезков и вертикальных соединений.
На них фаза нестационарна. Интегрируем по частям, используя
\[
\int e^{iNt^2}f(t)\,\dd t
 =\left.\frac{e^{iNt^2}f(t)}{2iNt}\right|_{\partial\gamma}
 -\frac1{2iN}\int e^{iNt^2}\left(\frac{f(t)}t\right)'\dd t.
\]
Повторение даёт экспоненциально малые граничные члены и остаток
$(2iN)^{-m}\int e^{iNt^2}F_m(t)\,\dd t=O(N^{-m})$.
На вертикальных частях используется следующая формула интегрирования по частям.
\end{myproof}
\manuscriptfigure{С. 43: срезающая функция $f=1$ около нуля; контур из внешних вещественных отрезков, двух вертикальных соединений и центрального отрезка под углом $\pi/4$.}
\begin{mylemma}
Пусть $[B,C]$ --- вертикальный отрезок, $u$ гладкая вдоль него, а $v$ аналитична в его окрестности.
Тогда
\[
\int_B^C u'(z)v(z)\,\dd z
 =-\int_B^C u(z)v'(z)\,\dd z+u(C)v(C)-u(B)v(B).
\]
\end{mylemma}
\begin{myproof}
Параметризуем отрезок: $z=x_0+iy$, $y\in[y_B,y_C]$, $\dd z=i\,\dd y$,
$\dd/\dd z=-i\,\dd/\dd y$. Обычное интегрирование по частям даёт
\[\begin{array}{@{}>{\displaystyle}r@{}>{\displaystyle}l@{}}
\int_B^C u'v\,\dd z
&=\int_{y_B}^{y_C}\frac{\dd}{\dd y}u(x_0+iy)v(x_0+iy)\,\dd y\\
&=[u(x_0+iy)v(x_0+iy)]_{y_B}^{y_C}
 -\int_{y_B}^{y_C}u(x_0+iy)\frac{\dd}{\dd y}v(x_0+iy)\,\dd y.
\end{array}\]
\end{myproof}
\begin{mylemma}
При тех же условиях на $f$
\[
\int_{-a}^a e^{iNt^2}tf(t)\,\dd t=O(N^{-m})\qquad\forall m.
\]
\end{mylemma}
\begin{myproof}
На деформированном центральном отрезке интеграл равен
$e^{i\pi/2}\int_{-\delta/\sqrt2}^{\delta/\sqrt2}e^{-Nx^2}x\,\dd x=0$
по нечётности. Остальные части оцениваются как в предыдущей лемме.
\end{myproof}
\begin{mylemma}
При тех же условиях
\[
\int_{-a}^a e^{iNt^2}t^2f(t)\,\dd t=O(N^{-3/2}).
\]
\end{mylemma}
\begin{myproof}
После той же деформации центральная часть по модулю не превосходит
\[
\int_{\mathbb R}e^{-Nx^2}x^2\,\dd x
 =N^{-3/2}\int_{\mathbb R}e^{-y^2}y^2\,\dd y.
\]
Остальные части имеют сколь угодно высокий порядок малости.
\end{myproof}
\begin{mytheorem}
Для $f\in C_0^\infty(-a,a)$
\[
\boxed{\int_{-a}^a e^{iNt^2}f(t)\,\dd t
 =\sqrt{\frac\pi N}e^{i\pi/4}f(0)+O(N^{-3/2}).}
\]
\end{mytheorem}
\begin{myproof}
Выбираем $\psi\in C_0^\infty(-a,a)$, $\psi=1$ на $[-\delta,\delta]$.
По лемме о нестационарной фазе
\[
\int_{-a}^a e^{iNt^2}f(t)\,\dd t
 =\int_{-a}^a e^{iNt^2}f(t)\psi(t)\,\dd t+O(N^{-m})\qquad\forall m.
\]
Разложим
$f(t)=f(0)+f'(0)t+\tfrac12f''(0)t^2+t^3h(t)$, где $h\in C^\infty$.
Три первых члена оцениваются предыдущими леммами.
Остаётся интеграл с $q(t)=t^3h(t)\psi(t)$.
Дважды интегрируя по частям, получаем
\[
\int_{-a}^a e^{iNt^2}q(t)\,\dd t
 =\frac1{(2iN)^2}\int_{-a}^a e^{iNt^2}
 \left[\frac1t\left(\frac{q(t)}t\right)'\right]'\dd t=O(N^{-2}).
\]
Граничные члены исчезают; подынтегральная амплитуда гладкая и ограниченная,
так как $q$ содержит множитель $t^3$.
\end{myproof}
\begin{myprop}
Для $f\in C_0^\infty(-a,a)$
\[
\int_{-a}^a e^{-iNt^2}f(t)\,\dd t
 =\sqrt{\frac\pi N}e^{-i\pi/4}f(0)+O(N^{-3/2}).
\]
\end{myprop}
\begin{myproof}
Применяем теорему к $\overline f$ и комплексно сопрягаем результат.
\end{myproof}
\begin{myremark}
Если $f\in C^\infty[0,a]$ и $f=0$ в окрестности $a$, то
\[
\int_0^a e^{iNt^2}f(t)\,\dd t
 =\frac12\sqrt{\frac\pi N}e^{i\pi/4}f(0)+O(N^{-1}).
\]
\end{myremark}

\subsection{Общая формула стационарной фазы}
\begin{mytheorem}
Пусть $\varphi,F\in C^\infty[a,b]$, $\varphi$ вещественна, все её стационарные точки
$c_k$ невырождены: $\varphi'(c_k)=0$, $\varphi''(c_k)\ne0$.
Главные вклады в $I(N)$ дают концы $a,b$ и точки $c_k$.
Если $\varphi'(a)\ne0$, то вклад левого конца
\[
I_a=-\frac{e^{iN\varphi(a)}F(a)}{iN\varphi'(a)}+O(N^{-2});
\]
если $\varphi'(b)\ne0$, то вклад правого конца
\[
I_b=\frac{e^{iN\varphi(b)}F(b)}{iN\varphi'(b)}+O(N^{-2}).
\]
Для $c_k\in(a,b)$
\[
\boxed{I_{c_k}=e^{iN\varphi(c_k)}
 \sqrt{\frac{2\pi}{N|\varphi''(c_k)|}}
 e^{i\pi\operatorname{sgn}\varphi''(c_k)/4}F(c_k)+O(N^{-3/2}).}
\]
Если $c_k=a$ или $b$, соответствующий главный член умножается на $1/2$, а остаток равен $O(N^{-1})$.
Интеграл равен сумме этих локальных вкладов.
% В формулировке на с. 45 у стационарных вкладов пропущен множитель e^{iN\varphi(c_k)};
% он явно присутствует в доказательстве на с. 46 и здесь восстановлен.
\end{mytheorem}
\begin{myproof}
Вблизи каждой $c_k$ вводим гладкую координату $t_k$:
\[
\varphi(x)=\varphi(c_k)+\frac{\varphi''(c_k)}2t_k(x)^2,
\quad t_k(c_k)=0,\quad t_k'(c_k)=1.
\]
Выбираем обратную функцию $\chi_k$ на $[-\delta_k,\delta_k]$,
$\chi_k(0)=c_k$, $\chi_k'(0)=1$.
Покрываем $[a,b]$ этими окрестностями, окрестностями концов и промежутками без стационарных точек.
Выбираем подчинённое гладкое разбиение единицы $\{\eta_j\}_{j=1}^M$:
\[
\sum_{j=1}^M\eta_j(x)=1\quad\text{на }[a,b],\qquad
I(N)=\sum_{j=1}^M\int_a^b e^{iN\varphi(x)}F(x)\eta_j(x)\,\dd x.
\]
Внутренние нестационарные части имеют порядок $O(N^{-m})$ для любого $m$.
Вклады концов следуют из интегрирования по частям.
Около внутренней стационарной точки, выбирая локальную срезку равной единице при $c_k$, имеем
\[
I_{c_k}=e^{iN\varphi(c_k)}\int_{-\delta_k}^{\delta_k}
 e^{iN\varphi''(c_k)t^2/2}
 F(\chi_k(t))\eta_k(\chi_k(t))\chi_k'(t)\,\dd t.
\]
Применяем теорему о квадратичной фазе с учётом знака $\varphi''(c_k)$.
В стационарном конце применяется её односторонний вариант.
\end{myproof}

J_\nu(z)=\sum_{k=0}^\infty\frac{(-1)^k}{k!\,\Gamma(k+\nu+1)}

J_n(z)=\frac1{2\pi}\int_{-\pi}^\pi e^{iz\sin\theta-in\theta}\,\dd\theta.

Для $z=N\to+\infty$ имеем $\varphi(\theta)=\sin\theta$, $F(\theta)=e^{-in\theta}$,

\subsection{Поведение волновой функции при больших временах}

\psi|_{t=0}=\psi_0(x).
\end{cases}

Преобразование Фурье

\widehat\psi(t,k)=\frac1{\sqrt{2\pi}}\int_{\mathbb R}e^{-ikx}\psi(t,x)\,\dd x

\section{Метод перевала}
Пусть
\[
I(N)=\int_C e^{N\varphi(z)}F(z)\,\dd z,\qquad C\subset\mathbb C,
\]
а $\varphi,F$ голоморфны в окрестности контура и области его допустимой деформации.
\subsection{Основная идея метода}
Разложение экспоненты имеет вид
$e^{N\varphi}=e^{N\operatorname{Re}\varphi}e^{iN\operatorname{Im}\varphi}$.
Нужно выбирать контур с возможно меньшим максимумом $\operatorname{Re}\varphi$.
Поскольку
\[
\Delta\varphi=(\partial_x-i\partial_y)(\partial_x+i\partial_y)\varphi=0,
\qquad\Delta\operatorname{Re}\varphi=0,
\]
вещественная часть непостоянной голоморфной функции не имеет внутреннего локального максимума.
Если два допустимо деформируемых контура $C_1,C_2$ соединяют $A$ и $B$, причём
\[
\max_{C_1}\operatorname{Re}\varphi=\operatorname{Re}\varphi(z_1)
 >\max_{C_2}\operatorname{Re}\varphi=\operatorname{Re}\varphi(z_2),
\]
то оценка по $C_2$ может быть экспоненциально точнее:
$|I(N)|=O(e^{N\operatorname{Re}\varphi(z_2)})$.
\begin{myexample}
Для $\varphi(z)=z^2$, $z=x+iy$,
\[
\varphi(z)=x^2-y^2+2ixy,\quad
\operatorname{Re}\varphi=x^2-y^2,\quad\operatorname{Im}\varphi=2xy.
\]
В точке $z_0=0$ имеем $\varphi'(z_0)=0$, $\varphi''(z_0)\ne0$.
По вещественной оси $\operatorname{Re}\varphi=x^2$, по мнимой $\operatorname{Re}\varphi=-y^2$;
поверхность $\operatorname{Re}\varphi$ имеет седло.
В общем случае
\[
\varphi(z)=\varphi(z_0)+\frac{\varphi''(z_0)}2(z-z_0)^2+O((z-z_0)^3).
\]
\end{myexample}
\manuscriptfigure{С. 47: два контура $C_1,C_2$ между $A,B$; седловая поверхность $\operatorname{Re}z^2$; оси и прямые $y=\pm x$, на которых $\operatorname{Re}z^2=0$.}
\begin{enumerate}
\item Если концы $A,B$ находятся по одну сторону перевала и
$\operatorname{Re}\varphi(B)>\operatorname{Re}\varphi(A)$, контур можно провести по $AM$ с
$\operatorname{Re}\varphi=\operatorname{Re}\varphi(A)$ и затем по $MB$ с
$\operatorname{Im}\varphi=\mathrm{const}$. Вклад $AM$ экспоненциально мал, на $MB$ применяется метод Лапласа.
\item Если $\operatorname{Re}\varphi(A),\operatorname{Re}\varphi(B)<\operatorname{Re}\varphi(z_0)$,
контур проводят по $AM_1$, $M_1M_2$, $M_2B$.
На внешних частях вещественная часть фазы постоянна и равна её значению в соответствующем конце;
на центральной части $\operatorname{Im}\varphi=\mathrm{const}$ и контур проходит через перевал $z_0$.
Внешние вклады экспоненциально малы, центральный вычисляется методом Лапласа.
\end{enumerate}
Если направление ориентированной касательной центрального контура при $z_0$ задано углом $\alpha$,
так что $z-z_0\sim s e^{i\alpha}$, $s\in\mathbb R$, и
$\varphi''(z_0)e^{2i\alpha}=-|\varphi''(z_0)|$, то
\[
\boxed{I(N)\sim\sqrt{\frac{2\pi}{N|\varphi''(z_0)|}}
 e^{N\varphi(z_0)}e^{i\alpha}F(z_0).}
\]
\manuscriptfigure{С. 48: два варианта деформации около перевала; в первом контур обходит седло с одной стороны, во втором проходит через него по линии постоянной мнимой части фазы.}
\begin{myremark}
При деформации нужно учитывать полюса $F$ и точки ветвления.
Пересечение полюса изменяет интеграл на соответствующий вычет, а ветвление требует согласованного выбора ветви и контура.
\end{myremark}
\subsection{Уравнение Эйри}
Рассмотрим уравнение
\[
w''(t)-tw(t)=0.
\]
Ищем решение в виде $w(t)=\int_\Gamma e^{tz}V(z)\,\dd z$. Тогда
\[\begin{array}{@{}>{\displaystyle}r@{}>{\displaystyle}l@{}}
w''-tw&=\int_\Gamma(e^{tz}z^2V-te^{tz}V)\,\dd z\\
&=\int_\Gamma e^{tz}(z^2V+V')\,\dd z-\left.e^{tz}V(z)\right|_{\partial\Gamma}.
\end{array}\]
Выбираем $V'=-z^2V$, то есть $V(z)=Ce^{-z^3/3}$, а концы контура уводим в сектора,
где $\operatorname{Re}z^3\to+\infty$ и граничные члены исчезают.
Получаем решения
\[
w(t)=\int_\Gamma e^{tz-z^3/3}\,\dd z.
\]
\manuscriptfigure{С. 48: три направления ухода контура на бесконечность: положительная вещественная полуось и лучи с аргументами $\pm2\pi/3$.}
Для контура из нижнего левого сектора к положительной вещественной полуоси найдём асимптотику при $t\to+\infty$.
Замена $z=\sqrt t\,\zeta$ даёт
\[
w(t)=\sqrt t\int_{\widetilde\Gamma}
 e^{t\sqrt t(\zeta-\zeta^3/3)}\,\dd\zeta,
\qquad N=t\sqrt t,\quad F=1,\quad\varphi(\zeta)=\zeta-\zeta^3/3.
\]
Пусть $\zeta=\xi+i\eta$. Тогда
\[
\operatorname{Re}\varphi=\xi-\frac{\xi^3}3+\xi\eta^2,
\qquad
\operatorname{Im}\varphi=\eta\left(1-\xi^2+\frac{\eta^2}3\right).
\]
Линии $\operatorname{Im}\varphi=0$ задаются $\eta=0$ или $\xi^2-\eta^2/3=1$;
асимптоты гипербол --- $\eta=\pm\sqrt3\,\xi$.
Стационарные точки $\zeta=\pm1$ имеют значения
$\varphi(-1)=-2/3$, $\varphi(1)=2/3$.
Для выбранного контура главный вклад даёт точка $\zeta=1$, где $\varphi''(1)=-2$:
\[
w(t)\sim\sqrt t\sqrt{\frac{2\pi}{2t\sqrt t}}e^{2t\sqrt t/3}.
\]
Итак,
\[
\boxed{w(t)=\frac{\sqrt\pi}{t^{1/4}}e^{\frac23t^{3/2}}
 \bigl(1+O(t^{-3/2})\bigr),\qquad t\to+\infty.}
\]
\manuscriptfigure{С. 49: линии $\operatorname{Im}(\zeta-\zeta^3/3)=0$ --- вещественная ось и две ветви гиперболы, седловые точки $\pm1$ и деформированный контур.}
\begin{myexe}
\begin{enumerate}
\item Найти асимптотику $w(t)$ при $t\to-\infty$.
\item Найти при $t\to+\infty$ асимптотику интеграла по другому контуру: из нижнего левого сектора в верхний левый сектор.
\end{enumerate}

\end{myexe}

\chapter{Интегральные уравнения}
Основные типы интегральных уравнений:
\[
\int_a^b K(t,s)f(s)\,\dd s=h(t)
\quad\text{--- уравнение Фредгольма первого рода},
\]
\[
f(t)+\int_a^b K(t,s)f(s)\,\dd s=h(t)
\quad\text{--- уравнение Фредгольма второго рода},
\]
\[
\int_a^t K(t,s)f(s)\,\dd s=h(t)
\quad\text{--- уравнение Вольтерра первого рода},
\]
\[
f(t)+\int_a^t K(t,s)f(s)\,\dd s=h(t)
\quad\text{--- уравнение Вольтерра второго рода}.
\]
\section{Линейные операторы в банаховых пространствах}
\subsection{Определения}
Пусть $X$ --- полное линейное нормированное пространство (банахово пространство).
\begin{myexample}
\[
X=C[a,b],\qquad\|f\|_X=\max_{[a,b]}|f(x)|;
\qquad
X=L_2(a,b),\qquad\|f\|_X=\left(\int_a^b|f(x)|^2\,\dd x\right)^{1/2}.
\]
\end{myexample}
\begin{mydef}
Оператор $A:X\to X$ называется линейным, если
$A(\alpha x+\beta y)=\alpha Ax+\beta Ay$.
Он называется ограниченным, если существует $C$ такое, что
$\|Ax\|_X\le C\|x\|_X$ для всех $x\in X$.
Его норма
\[
\|A\|=\sup_{x\ne0}\frac{\|Ax\|_X}{\|x\|_X},\qquad
\|Ax\|_X\le\|A\|\|x\|_X.
\]
\end{mydef}
\begin{mylemma}
Ограниченный линейный оператор непрерывен.
\end{mylemma}
\begin{myproof}
Если $x_n\to x$, то $\|Ax_n-Ax\|_X\le\|A\|\|x_n-x\|_X\to0$.
\end{myproof}
\begin{myexe}
Доказать, что непрерывный линейный оператор ограничен.
\end{myexe}
Обозначим через $B(X)$ пространство всех ограниченных линейных операторов в $X$.
Сложение определяется поточечно: $(A_1+A_2)x=A_1x+A_2x$.
Операторная норма обладает свойствами
\[
\|A\|\ge0,\quad\|A\|=0\Longleftrightarrow A=0,\quad
\|\alpha A\|=|\alpha|\|A\|,\quad\|A_1+A_2\|\le\|A_1\|+\|A_2\|.
\]
Последнее следует из
$\|(A_1+A_2)x\|_X\le\|A_1x\|_X+\|A_2x\|_X
\le(\|A_1\|+\|A_2\|)\|x\|_X$.
\begin{myexe}
Доказать полноту $B(X)$ в операторной норме.
\end{myexe}
Произведение операторов --- композиция: $(A_1A_2)x=A_1(A_2x)$.
Имеем
\[
\|A_1A_2\|\le\|A_1\|\|A_2\|,\quad A(BC)=(AB)C,\quad
(A+B)C=AC+BC.
\]
Определяем $A^m=A\cdots A$ ($m$ множителей), $A^0=I$.
\begin{mydef}
Если $BA=AB=I$, то $B$ называется обратным к $A$ и обозначается $A^{-1}$.
Для обратимого $A$ определены целые степени, причём $A^{n+m}=A^nA^m$.
\end{mydef}

\subsection{Ряды в банаховых пространствах}
\begin{mydef}
Ряд $\sum_{k=1}^\infty x_k$ сходится к $y\in X$, если
$\|\sum_{k=1}^n x_k-y\|_X\to0$ при $n\to\infty$.
\end{mydef}
\begin{mylemma}
Если $\sum_{k=1}^\infty\|x_k\|_X<\infty$, то ряд $\sum_{k=1}^\infty x_k$ сходится.
\end{mylemma}
\begin{myproof}
Для $\varepsilon>0$ выбираем $N$ так, что $\sum_{k=N}^\infty\|x_k\|_X<\varepsilon$.
Если $n>m>N$, то для $S_n=\sum_{k=1}^n x_k$
\[
\|S_n-S_m\|_X=\left\|\sum_{k=m+1}^n x_k\right\|_X
 \le\sum_{k=m+1}^n\|x_k\|_X<\varepsilon.
\]
Последовательность частичных сумм фундаментальна; полнота $X$ даёт её предел.
\end{myproof}
\begin{myremark}
$\|\sum_{k=1}^\infty x_k\|_X\le\sum_{k=1}^\infty\|x_k\|_X$.
Это применимо и к рядам операторов в $B(X)$.
\end{myremark}
\begin{mylemma}
Пусть $A\in B(X)$ и $\sum_{k=0}^\infty\|A^k\|<\infty$.
Тогда $(I-A)^{-1}\in B(X)$ и
\[
\boxed{(I-A)^{-1}=\sum_{k=0}^\infty A^k.}
\]
Это ряд Неймана.
\end{mylemma}
\begin{myproof}
Положим $B=\sum_{k=0}^\infty A^k=I+A+A^2+\cdots$.
В силу сходимости в операторной норме можно умножать ряд на $A$:
$AB=A+A^2+\cdots=B-I$, $BA=B-I$.
Следовательно, $(I-A)B=B(I-A)=I$.
\end{myproof}
\begin{myprop}
Если $\|A\|<1$, то $(I-A)^{-1}$ существует и
\[
\|(I-A)^{-1}\|\le\frac1{1-\|A\|}.
\]
\end{myprop}
\begin{myproof}
$\|A^k\|\le\|A\|^k$, поэтому сумма норм ограничена геометрическим рядом.
\end{myproof}
\begin{myprop}
Если $A\in B(X)$, $\lambda\in\mathbb C$, $|\lambda|>\|A\|$, то $(A-\lambda I)^{-1}\in B(X)$.
\end{myprop}
\begin{myproof}
$\|A/\lambda\|<1$ и
$A-\lambda I=-\lambda(I-A/\lambda)$, откуда
$(A-\lambda I)^{-1}=-\lambda^{-1}(I-A/\lambda)^{-1}$.
\end{myproof}

\subsection{Уравнения Вольтерра}
Пусть $X=C[a,b]$, $h\in C[a,b]$, а ядро $K(t,s)$ непрерывно в треугольнике $a\le s\le t\le b$.
Рассмотрим
\[
f(t)-\int_a^t K(t,s)f(s)\,\dd s=h(t).
\]
Обозначая тем же символом интегральный оператор,
\[
(Kf)(t)=\int_a^t K(t,s)f(s)\,\dd s,\qquad K:X\to X,
\]
переписываем уравнение как $(I-K)f=h$.
\begin{mylemma}
Если $M=\max_{a\le s\le t\le b}|K(t,s)|$, то
\[
|(K^jf)(t)|\le\frac{M^j(t-a)^j}{j!}\|f\|_C,\qquad j\ge0.
\]
\end{mylemma}
\begin{myproof}
При $j=0$ утверждение очевидно. Индукционный переход:
\[\begin{array}{@{}>{\displaystyle}r@{}>{\displaystyle}l@{}}
|(K^{j+1}f)(t)|
&\le M\int_a^t\frac{M^j(s-a)^j}{j!}\|f\|_C\,\dd s\\
&=\frac{M^{j+1}(t-a)^{j+1}}{(j+1)!}\|f\|_C.
\end{array}\]
\end{myproof}
\begin{mytheorem}[unbreakable]
При непрерывном ядре $K$ в указанном треугольнике для каждого $h\in C[a,b]$ существует единственное
$f\in C[a,b]$, удовлетворяющее уравнению Вольтерра второго рода. При этом
\[
\boxed{\|f\|_C\le e^{M(b-a)}\|h\|_C.}
\]
\end{mytheorem}
\begin{myproof}
Из леммы $\|K^j\|\le M^j(b-a)^j/j!$, поэтому
$\sum_{j=0}^\infty\|K^j\|\le e^{M(b-a)}$.
Ряд Неймана даёт $(I-K)^{-1}=I+K+K^2+\cdots$ и нужную оценку.
\end{myproof}
\begin{myremark}
Решение представляется рядом
\[\begin{array}{@{}>{\displaystyle}r@{}>{\displaystyle}l@{}}
f(t)=h(t)&+\int_a^t K(t,s)h(s)\,\dd s\\
&+\int_a^t K(t,t_1)\int_a^{t_1}K(t_1,s)h(s)\,\dd s\,\dd t_1+\cdots.
\end{array}\]
\end{myremark}
\begin{myexe}
Пусть $h\in C[0,\infty)$, $|h(t)|\le L$, ядро непрерывно на $0\le s\le t$ и $|K(t,s)|\le M$.
Доказать существование и единственность $f\in C[0,\infty)$ для
\[
f(t)-\int_0^t K(t,s)f(s)\,\dd s=h(t),\qquad |f(t)|\le Le^{Mt}.
\]
\end{myexe}
\begin{myremark}
Для уравнения Вольтерра первого рода
$\int_a^t K(t,s)f(s)\,\dd s=h(t)$ необходимо $h(a)=0$.
Если $K,h\in C^1$, дифференцирование даёт
\[
K(t,t)f(t)+\int_a^t\partial_tK(t,s)f(s)\,\dd s=h'(t).
\]
При $K(t,t)>0$ получаем уравнение второго рода:
\[
f(t)+\int_a^t\frac{\partial_tK(t,s)}{K(t,t)}f(s)\,\dd s
 =\frac{h'(t)}{K(t,t)}.
\]
\end{myremark}

\section{Гильбертовы пространства}
\subsection{Определения}
\begin{mydef}
Гильбертово пространство $H$ --- полное комплексное линейное пространство со скалярным произведением.
В рассматриваемом случае оно бесконечномерно. Приняты соглашения
\[
(x,y)=\overline{(y,x)},\qquad (x,x)\ge0,\quad (x,x)=0\Longleftrightarrow x=0,
\]
\[
(\alpha x_1+\beta x_2,y)=\alpha(x_1,y)+\beta(x_2,y),\qquad
\|x\|_H=\sqrt{(x,x)}.
\]
\end{mydef}
\begin{myexe}
Доказать неравенство Коши--Буняковского: $|(x,y)|\le\|x\|_H\|y\|_H$.
\end{myexe}
Если $x_n\to x$, $y_n\to y$, то $(x_n,y_n)\to(x,y)$, поскольку
\[
|(x_n,y_n)-(x,y)|
\le\|x_n-x\|_H\|y_n\|_H+\|x\|_H\|y_n-y\|_H\to0.
\]
Здесь сходящаяся последовательность $y_n$ ограничена.
\begin{mydef}
Пространство $H$ сепарабельно, если в нём существует счётное плотное множество
$Y=\{y_j\}_{j=1}^\infty$.
\end{mydef}
\begin{myremark}
Если $z\perp y_j$ для всех $j$, то $z=0$.
Действительно, из плотности выбирается $y_{j_k}\to z$; тогда
$0=(y_{j_k},z)\to\|z\|_H^2$.
\end{myremark}
\begin{myexe}
Доказать, что любое подпространство сепарабельного гильбертова пространства сепарабельно.
\end{myexe}
\begin{mylemma}
Пусть $\{\varphi_k\}_{k=1}^\infty$ --- ортонормированная система:
$(\varphi_k,\varphi_j)=\delta_{kj}$. Тогда для каждого $x\in H$ при $c_k=(x,\varphi_k)$
ряд $\sum_{k=1}^\infty c_k\varphi_k$ сходится в норме $H$.
\end{mylemma}
\begin{myproof}
Для $S_N=\sum_{k=1}^N c_k\varphi_k$ имеем
\[
\|S_N\|_H^2=\sum_{k=1}^N|c_k|^2=(x,S_N)
\le\|x\|_H\|S_N\|_H.
\]
Поэтому $\sum_{k=1}^N|c_k|^2\le\|x\|_H^2$, и, переходя к пределу, получаем неравенство Бесселя
\[
\boxed{\sum_{k=1}^\infty|c_k|^2\le\|x\|_H^2.}
\]
При $n>m$
\[
\|S_n-S_m\|_H^2=\sum_{k=m+1}^n|c_k|^2\to0.
\]
Полнота $H$ даёт сходимость ряда.
\end{myproof}
\begin{myremark}
Сходимость этого ряда не обязана быть абсолютной: ряд
$\sum\|c_k\varphi_k\|_H=\sum|c_k|$ может расходиться.
\end{myremark}
\begin{mylemma}
В сепарабельном гильбертовом пространстве существует счётный ортонормированный базис.
\end{mylemma}
\begin{myproof}
Берём счётное плотное множество $\{y_j\}_{j=1}^\infty$ и применяем процесс ортогонализации,
пропуская векторы, лежащие в линейной оболочке предыдущих:
\[
\varphi_1=\frac{y_1}{\|y_1\|_H},\quad
\varphi_2=\frac{y_2-(y_2,\varphi_1)\varphi_1}
 {\|y_2-(y_2,\varphi_1)\varphi_1\|_H},
\]
\[
\varphi_3=\frac{y_3-(y_3,\varphi_1)\varphi_1-(y_3,\varphi_2)\varphi_2}
 {\|y_3-(y_3,\varphi_1)\varphi_1-(y_3,\varphi_2)\varphi_2\|_H},\quad\ldots.
\]
Получаем $(\varphi_k,\varphi_j)=\delta_{kj}$ и совпадение линейных оболочек выбранных начальных систем.
Для $x\in H$ положим $c_k=(x,\varphi_k)$, $\widetilde x=\sum_{k=1}^\infty c_k\varphi_k$.
По предыдущей лемме этот ряд сходится, а
$(x-\widetilde x,\varphi_j)=c_j-c_j=0$ для всех $j$.
Значит, $x-\widetilde x\perp y_j$ для всех $j$. Из плотности следует $x=\widetilde x$.
Таким образом, система полна и является ортонормированным базисом.
\end{myproof}
\begin{myremark}
\begin{enumerate}
\item Существование счётного ортонормированного базиса влечёт сепарабельность:
счётное плотное множество образуют конечные суммы
\[
\left\{\sum_{k=1}^N(\alpha_k+i\beta_k)\varphi_k:
 \alpha_k,\beta_k\in\mathbb Q,\ N\in\mathbb N\right\}.
\]
\item Если $H_1,H_2$ --- сепарабельные бесконечномерные гильбертовы пространства с ортонормированными базисами
$\{\varphi_k\}$, $\{\psi_k\}$, то отображение
\[
H_1\to H_2,\qquad x=\sum_{k=1}^\infty c_k\varphi_k
 \longmapsto y=\sum_{k=1}^\infty c_k\psi_k
\]
является изоморфизмом гильбертовых пространств.
\end{enumerate}
\end{myremark}
